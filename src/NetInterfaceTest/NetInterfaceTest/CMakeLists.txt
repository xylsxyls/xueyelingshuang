cmake_minimum_required(VERSION 3.15)
if(BIT STREQUAL "wasm64")
    string(APPEND CMAKE_C_FLAGS " -sMEMORY64=1")
    string(APPEND CMAKE_CXX_FLAGS " -sMEMORY64=1")
    string(APPEND CMAKE_EXE_LINKER_FLAGS " -sMEMORY64=1")
endif()
include("${CMAKE_CURRENT_SOURCE_DIR}/../../PlatformTest/PlatformTest/src/PlatformTest.cmake")
if(NOT CMAKE_TOOLCHAIN_FILE)
    if(BIT MATCHES "^(32|64)$" AND CMAKE_HOST_SYSTEM_NAME STREQUAL "Linux"
        AND (NOT CMAKE_SYSTEM_NAME OR CMAKE_SYSTEM_NAME STREQUAL "Linux"))
        string(APPEND CMAKE_CXX_FLAGS_INIT " -m${BIT}")
    elseif((CMAKE_HOST_APPLE OR CMAKE_SYSTEM_NAME MATCHES "^(Darwin|iOS)$") AND NOT CMAKE_OSX_ARCHITECTURES)
        if(BIT STREQUAL "arm64")
            set(CMAKE_OSX_ARCHITECTURES arm64)
        elseif(BIT STREQUAL "64")
            set(CMAKE_OSX_ARCHITECTURES x86_64)
        endif()
    else()
        platform_test_select_linux_arm()
    endif()
endif()
project(NetInterfaceTest LANGUAGES CXX)
# 显式工具链或复用的构建目录也须与所选 SDK 架构相符。
if(BIT AND CMAKE_SYSTEM_NAME MATCHES "^(Linux|Darwin|iOS|Android)$")
    if(APPLE AND NOT BIT MATCHES "^(64|arm64)$")
        message(FATAL_ERROR "Apple SDKs require BIT=64 or arm64")
    endif()
    set(cpu_32 "defined(__i386__) && __SIZEOF_POINTER__ == 4")
    set(cpu_64 "defined(__x86_64__) && __SIZEOF_POINTER__ == 8")
    set(cpu_arm32 "defined(__arm__) && __SIZEOF_POINTER__ == 4")
    set(cpu_arm64 "(defined(__aarch64__) || defined(__arm64__)) && __SIZEOF_POINTER__ == 8")
    if(NOT DEFINED cpu_${BIT})
        message(FATAL_ERROR "Unsupported native BIT=${BIT}")
    endif()
    include(CheckCXXSourceCompiles)
    unset(NET_INTERFACE_TEST_ABI_MATCHES CACHE)
    check_cxx_source_compiles("#if !(${cpu_${BIT}})\n#error Wrong SDK target\n#endif\nint main() { return 0; }" NET_INTERFACE_TEST_ABI_MATCHES)
    if(NOT NET_INTERFACE_TEST_ABI_MATCHES)
        message(FATAL_ERROR "Compiler architecture disagrees with BIT=${BIT}")
    endif()
endif()
include(CTest)
if(MSVC)
    set(vendors zstd$<$<CONFIG:Debug>:d>.lib libprotobuf-lite$<$<CONFIG:Debug>:d>.lib mbedtls$<$<CONFIG:Debug>:d>.lib)
    set(system_libraries ws2_32 iphlpapi psapi userenv bcrypt shell32 ole32 oleaut32 uuid gdi32 user32 dbghelp advapi32 crypt32 secur32 normaliz wldap32 winmm)
else()
    set(vendors libzstd.a libprotobuf-lite.a libmbedtls.a)
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
        # 测试直接链接 SDK 归档，在这里补齐 NetInterface 的 Linux 系统链接项。
        list(APPEND system_libraries rt atomic)
    elseif(APPLE)
        list(APPEND system_libraries "-framework Security")
    endif()
endif()
build_sdk_test(NetInterfaceTest PROJECTS NetInterface NetChannel CStringManager Compress ProtoMessageManager Variant CEncodeDecode LockFreeQueue LockFreeHashMap
    ARCHIVES ${vendors} SYSTEM_LIBRARIES ${system_libraries})
