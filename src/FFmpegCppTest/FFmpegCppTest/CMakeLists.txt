cmake_minimum_required(VERSION 3.15)
if(BIT STREQUAL "wasm64")
    string(APPEND CMAKE_C_FLAGS_INIT " -sMEMORY64=1")
    string(APPEND CMAKE_CXX_FLAGS_INIT " -sMEMORY64=1")
    string(APPEND CMAKE_EXE_LINKER_FLAGS_INIT " -sMEMORY64=1")
endif()
project(FFmpegCppTest LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(root "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
include("${root}/src/PlatformTest/PlatformTest/src/PlatformTest.cmake")
if(NOT SDK_ROOT)
    set(SDK_ROOT "$ENV{CLOUD_REBUILD_SDK_ROOT}")
endif()
if(NOT SDK_ROOT)
    set(SDK_ROOT "${root}")
endif()
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:DebugDLL>")
endif()
add_executable(FFmpegCppTest FFmpegCppTest.cpp)
platform_test_link(FFmpegCppTest "${SDK_ROOT}")
target_include_directories(FFmpegCppTest PRIVATE "${SDK_ROOT}/include")
if(EMSCRIPTEN)
    platform_test_wasm_headers(FFmpegCppTest "${SDK_ROOT}" FFmpegCpp)
    math(EXPR wasm_bits "${CMAKE_SIZEOF_VOID_P} * 8")
    if(BIT AND NOT BIT STREQUAL "wasm${wasm_bits}")
        message(FATAL_ERROR "BIT disagrees with the Emscripten pointer size")
    endif()
    if(DEBUGRELEASE STREQUAL "debug" OR CMAKE_BUILD_TYPE STREQUAL "Debug")
        set(suffix d)
    endif()
    target_compile_definitions(FFmpegCppTest PRIVATE STATIC_LIB)
    target_compile_options(FFmpegCppTest PRIVATE -fexceptions)
    # Match the stack provisioned by the published FFmpegCpp runtime for codec calls.
    target_link_options(FFmpegCppTest PRIVATE -fexceptions -sALLOW_MEMORY_GROWTH=1 -sSTACK_SIZE=1048576)
    target_link_libraries(FFmpegCppTest PRIVATE "${SDK_ROOT}/lib/FFmpegCpp${suffix}.a")
    if(NOT LIBRARIESRELY)
        file(STRINGS "${SDK_ROOT}/include/FFmpegCpp/FFmpegCppWasmSDK.json" prepared_links
            REGEX "\"librariesrely\": *\"")
        string(REGEX MATCH "\"librariesrely\": *\"([^\"]+)\"" matched "${prepared_links}")
        if(NOT matched)
            message(FATAL_ERROR "Prepare FFmpegCpp and its matching vendor archives first")
        endif()
        string(REPLACE "\\;" ";" LIBRARIESRELY "${CMAKE_MATCH_1}")
    endif()
    foreach(dependency IN LISTS LIBRARIESRELY)
        if(IS_ABSOLUTE "${dependency}")
            target_link_libraries(FFmpegCppTest PRIVATE "${dependency}")
        else()
            target_link_libraries(FFmpegCppTest PRIVATE "${SDK_ROOT}/lib/${dependency}")
        endif()
    endforeach()
else()
    platform_test_sdk_library(prepared_library FFmpegCpp)
    if(MSVC)
        target_link_directories(FFmpegCppTest PRIVATE "${SDK_ROOT}/lib")
    endif()
    target_link_libraries(FFmpegCppTest PRIVATE "${prepared_library}")
    if(UNIX AND NOT LIBRARIESRELY AND EXISTS "${SDK_ROOT}/lib/libavformat.a")
        # 多配置 SDK 可同时含动态库和静态库，仅在本配置选中静态 FFmpegCpp 时补传递依赖。
        set(static_archives "${SDK_ROOT}/lib/libFFmpegCpp.a;${SDK_ROOT}/lib/libFFmpegCppd.a")
        foreach(component avformat avcodec swscale swresample avutil SoundTouchDLL openh264)
            target_link_libraries(FFmpegCppTest PRIVATE
                "$<$<IN_LIST:${prepared_library},${static_archives}>:${SDK_ROOT}/lib/lib${component}.a>")
        endforeach()
    endif()
    foreach(dependency IN LISTS LIBRARIESRELY)
        if(IS_ABSOLUTE "${dependency}")
            target_link_libraries(FFmpegCppTest PRIVATE "${dependency}")
        else()
            target_link_libraries(FFmpegCppTest PRIVATE "${SDK_ROOT}/lib/${dependency}")
        endif()
    endforeach()
    if(UNIX)
        find_package(Threads REQUIRED)
        target_link_libraries(FFmpegCppTest PRIVATE Threads::Threads)
        if(ANDROID)
            target_link_libraries(FFmpegCppTest PRIVATE log m)
        elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
            target_link_libraries(FFmpegCppTest PRIVATE ${CMAKE_DL_LIBS} m)
        endif()
    endif()
endif()
enable_testing()
add_library_platform_test(FFmpegCppTest TAKES_ARGS)
