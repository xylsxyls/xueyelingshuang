# 链接探测不能证明无锁性质，运行时仍验证所有必需宽度
include(CheckCXXSourceCompiles)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/LockFreeMapAtomicProbe.cpp")
function(lock_free_map_test_atomic_libraries output)
    if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
        # 保留设备工具链的只编译和签名策略，由真实目标测试完成验证
        set(${output} "" PARENT_SCOPE)
        return()
    endif()
    set(CMAKE_TRY_COMPILE_TARGET_TYPE EXECUTABLE)
    file(READ "${CMAKE_CURRENT_SOURCE_DIR}/LockFreeMapAtomicProbe.cpp" atomic_probe)
    # 使用新结果名，避免迁移前仅检查 uint64_t 的缓存被当作完整检查
    check_cxx_source_compiles("${atomic_probe}" LOCK_FREE_MAP_TEST_ALL_ATOMICS_DIRECT)
    if(LOCK_FREE_MAP_TEST_ALL_ATOMICS_DIRECT)
        set(${output} "" PARENT_SCOPE)
        return()
    endif()
    list(APPEND CMAKE_REQUIRED_LIBRARIES atomic)
    check_cxx_source_compiles("${atomic_probe}" LOCK_FREE_MAP_TEST_ALL_ATOMICS_LIBRARY)
    if(NOT LOCK_FREE_MAP_TEST_ALL_ATOMICS_LIBRARY)
        message(FATAL_ERROR "目标无法链接指针、size_t、uint32_t、uint64_t 原子操作")
    endif()
    set(${output} atomic PARENT_SCOPE)
endfunction()
lock_free_map_test_atomic_libraries(atomic_library)

foreach(target IN LISTS test_targets)
    set_target_properties(${target} PROPERTIES DEBUG_POSTFIX d)
    target_include_directories(${target} PRIVATE "${LOCK_FREE_MAP_TEST_SOURCE}" "${LOCK_FREE_MAP_TEST_SDK}")
    target_compile_definitions(${target} PRIVATE STATIC_LIB)
    if(atomic_library)
        target_link_libraries(${target} PRIVATE ${atomic_library})
    endif()
    if(TARGET Threads::Threads)
        target_link_libraries(${target} PRIVATE Threads::Threads)
    endif()
    if(MSVC)
        target_compile_options(${target} PRIVATE /EHsc /W4)
        # 共存用例只编译信号量公开类型，不调用其实现或消费共享 SDK 二进制
        target_link_options(${target} PRIVATE /NODEFAULTLIB:Semaphore.lib /NODEFAULTLIB:Semaphored.lib)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra)
        if(EMSCRIPTEN AND LOCK_FREE_MAP_TEST_WASM_NATIVE_EXCEPTIONS)
            target_compile_options(${target} PRIVATE -fwasm-exceptions)
            target_link_options(${target} PRIVATE -fwasm-exceptions)
        else()
            target_compile_options(${target} PRIVATE -fexceptions)
            if(EMSCRIPTEN)
                target_link_options(${target} PRIVATE -fexceptions)
            endif()
        endif()
    endif()
    if(WIN32)
        target_link_libraries(${target} PRIVATE psapi)
    endif()
    if(MINGW)
        # 故障注入替换全局分配器，静态链接 C++ 运行库以避免跨 DLL 混用分配与释放
        target_link_options(${target} PRIVATE -static-libgcc -static-libstdc++)
    endif()
    if(EMSCRIPTEN)
        target_link_options(${target} PRIVATE -sABORTING_MALLOC=0 -sEXIT_RUNTIME=1)
        if(LOCK_FREE_MAP_TEST_NODE_FILESYSTEM)
            target_link_options(${target} PRIVATE -sNODERAWFS=1)
        endif()
        if(LOCK_FREE_MAP_TEST_FIXED_HEAP)
            target_compile_definitions(${target} PRIVATE LOCK_FREE_MAP_TEST_FIXED_HEAP)
            target_link_options(${target} PRIVATE -sALLOW_MEMORY_GROWTH=0 -sINITIAL_MEMORY=16777216)
        else()
            target_link_options(${target} PRIVATE -sALLOW_MEMORY_GROWTH=1)
        endif()
        if(LOCK_FREE_MAP_TEST_PTHREADS)
            target_compile_options(${target} PRIVATE -pthread)
            target_link_options(${target} PRIVATE -pthread -sPTHREAD_POOL_SIZE=20 -sPROXY_TO_PTHREAD=1)
        endif()
    endif()
endforeach()
