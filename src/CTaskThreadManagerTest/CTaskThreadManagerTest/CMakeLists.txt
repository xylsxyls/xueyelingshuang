#设置CMAKE最低版本
CMAKE_MINIMUM_REQUIRED(VERSION 3.15)
get_filename_component(root "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
include("${root}/src/PlatformTest/PlatformTest/src/PlatformTest.cmake")
include("${root}/common/linux/LinuxArmAtomics.cmake")
platform_test_select_linux_arm()
option(CTTM_WASM_SMOKE "Build the standalone Emscripten pthread smoke test" OFF)
option(CTTM_IN_PROCESS_SMOKE "Build the single-process test for a native or mobile SDK" OFF)
option(CTTM_IN_PROCESS_CASES "Build registered test cases for a single-process host" OFF)
if(CTTM_WASM_SMOKE OR CTTM_IN_PROCESS_SMOKE OR CTTM_IN_PROCESS_CASES)
    cmake_minimum_required(VERSION 3.15)
    project(CTaskWasmSmoke LANGUAGES CXX)
    if(CTTM_WASM_SMOKE AND NOT EMSCRIPTEN)
        message(FATAL_ERROR "CTTM_WASM_SMOKE requires Emscripten")
    endif()
    if(NOT CTTM_INCLUDE_ROOT OR NOT CTTM_LIBRARY_ROOT)
        message(FATAL_ERROR "Set CTTM_INCLUDE_ROOT and CTTM_LIBRARY_ROOT to the matched target SDK")
    endif()
    if(CTTM_IN_PROCESS_CASES)
        set(test_target CTaskThreadCases)
        file(GLOB case_sources CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/*.cpp")
        list(REMOVE_ITEM case_sources "${CMAKE_CURRENT_SOURCE_DIR}/CTaskWasmSmoke.cpp")
        add_executable(${test_target} ${case_sources})
        target_compile_definitions(${test_target} PRIVATE CTTM_TEST_SINGLE_PROCESS)
    else()
        set(test_target CTaskWasmSmoke)
        add_executable(${test_target} CTaskWasmSmoke.cpp CTaskWasmSmoke.h)
    endif()
    set_target_properties(${test_target} PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES)
    target_compile_definitions(${test_target} PRIVATE STATIC_LIB)
    target_include_directories(${test_target} PRIVATE "${CTTM_INCLUDE_ROOT}")
    set(test_components CTaskThreadManager Semaphore LockFreeQueue LockFreeHashMap LockFreeMap)
    if(CTTM_IN_PROCESS_CASES)
        list(APPEND test_components PlatformTest)
    endif()
    # SDK 可以只发布一种配置。多配置生成器在构建对应目标时检查缺件，避免 Debug 误链 Release。
    foreach(component IN LISTS test_components)
        foreach(variant IN ITEMS release debug)
            set(basename "${component}")
            if(variant STREQUAL "debug")
                string(APPEND basename d)
            endif()
            unset(archive CACHE)
            find_file(archive NAMES "${basename}.lib" "${basename}.a" "lib${basename}.a"
                PATHS "${CTTM_LIBRARY_ROOT}" NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH)
            set(${variant}_archive "${archive}")
            unset(archive CACHE)
        endforeach()
        if(NOT release_archive AND NOT debug_archive AND NOT component MATCHES "^(CTaskThreadManager|Semaphore|PlatformTest)$")
            continue()
        endif()
        if(CMAKE_CONFIGURATION_TYPES)
            if(NOT debug_archive)
                set(debug_archive "${CTTM_LIBRARY_ROOT}/${component}d${CMAKE_STATIC_LIBRARY_SUFFIX}")
            endif()
            if(NOT release_archive)
                set(release_archive "${CTTM_LIBRARY_ROOT}/${component}${CMAKE_STATIC_LIBRARY_SUFFIX}")
            endif()
            target_link_libraries(${test_target} PRIVATE "$<$<CONFIG:Debug>:${debug_archive}>$<$<NOT:$<CONFIG:Debug>>:${release_archive}>")
        else()
            string(TOLOWER "${CMAKE_BUILD_TYPE}" build_type)
            if(build_type STREQUAL "debug" OR (NOT build_type AND DEBUGRELEASE STREQUAL "debug"))
                set(selected_archive "${debug_archive}")
            else()
                set(selected_archive "${release_archive}")
            endif()
            if(NOT selected_archive)
                message(FATAL_ERROR "Missing ${component} for the selected configuration under ${CTTM_LIBRARY_ROOT}")
            endif()
            target_link_libraries(${test_target} PRIVATE "${selected_archive}")
        endif()
    endforeach()
    if(EMSCRIPTEN)
        set_target_properties(${test_target} PROPERTIES SUFFIX ".js")
        target_compile_options(${test_target} PRIVATE -pthread -fexceptions)
        if(CTTM_IN_PROCESS_CASES)
            target_link_options(${test_target} PRIVATE -sDEFAULT_PTHREAD_STACK_SIZE=1048576 -sSTACK_SIZE=1048576 -sSTACK_OVERFLOW_CHECK=2)
        endif()
        set(CTTM_WASM_WORKERS 4 CACHE STRING "Prestarted host workers, size for the chosen case")
        target_link_options(${test_target} PRIVATE -pthread -fexceptions -sPROXY_TO_PTHREAD -sPTHREAD_POOL_SIZE=${CTTM_WASM_WORKERS} -sENVIRONMENT=node -sEXIT_RUNTIME=1 -sINITIAL_MEMORY=67108864 -sALLOW_MEMORY_GROWTH=1)
    else()
        find_package(Threads REQUIRED)
        target_link_libraries(${test_target} PRIVATE Threads::Threads)
        target_linux_arm_atomics(${test_target} PRIVATE)
        if(WIN32)
            target_link_libraries(${test_target} PRIVATE psapi advapi32 user32 ole32 oleaut32 uuid)
        endif()
        if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
            set_target_properties(${test_target} PROPERTIES MACOSX_BUNDLE YES)
        endif()
    endif()
    return()
endif()

SET(CMAKE_CXX_STANDARD 11)
if(NOT DEFINED DEBUGRELEASE)
    SET(DEBUGRELEASE release)
endif()
if(NOT DEFINED BIT)
    SET(BIT native)
endif()

#SET(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -finput-charset='gb2312'")
#SET(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -finput-charset='gb2312'")

if("${DEBUGRELEASE}" STREQUAL "debug")
#设置项目名称
SET(PROJECT_NAME CTaskThreadManagerTestd)
SET(CMAKE_CXX_FLAGS_DEBUG "$ENV{CXXFLAGS} -Wl,-rpath=. -O0 -g -ggdb ")
SET(CMAKE_BUILD_TYPE "Debug")
elseif("${DEBUGRELEASE}" STREQUAL "release")
#设置项目名称
SET(PROJECT_NAME CTaskThreadManagerTest)
SET(CMAKE_CXX_FLAGS_RELEASE "$ENV{CXXFLAGS} -Wl,-rpath=. -O2")
SET(CMAKE_BUILD_TYPE "Release")
endif()

if("${BIT}" STREQUAL "native" OR CMAKE_TOOLCHAIN_FILE OR (BIT MATCHES "^arm(32|64)$" AND (CMAKE_HOST_SYSTEM_NAME STREQUAL "Linux" OR CMAKE_SYSTEM_NAME STREQUAL "Linux")))
    MESSAGE(STATUS "using the architecture of the selected compiler/toolchain")
elseif("${BIT}" STREQUAL "32")
    MESSAGE(STATUS "using 32bits (x86)")
    SET(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -m32")
    SET(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -m32")
elseif("${BIT}" STREQUAL "64")
    MESSAGE(STATUS "using 64bits (x86_64)")
    SET(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -m64")
    SET(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -m64")
else()
    MESSAGE(FATAL_ERROR "Unknown BIT value: ${BIT}. Please set BIT to 'native', '32', '64', 'arm32', or 'arm64'")
endif()

#建立项目
PROJECT(${PROJECT_NAME} LANGUAGES CXX)

if(NOT SDK_ROOT)
    set(SDK_ROOT "$ENV{CLOUD_REBUILD_SDK_ROOT}")
endif()
if(NOT SDK_ROOT)
    set(SDK_ROOT "${root}")
    set(EXECUTABLE_OUTPUT_PATH "${root}/lib")
endif()

#获取代码，在项目中，将所有代码都放在src文件夹中
AUX_SOURCE_DIRECTORY(${PROJECT_SOURCE_DIR}/ DIR_SRCS)
# Wasm 检查有独立 main，只通过 CTTM_WASM_SMOKE 目标构建。
foreach(TEST_SOURCE ${DIR_SRCS})
    get_filename_component(TEST_SOURCE_NAME "${TEST_SOURCE}" NAME)
    if(TEST_SOURCE_NAME STREQUAL "CTaskWasmSmoke.cpp")
        list(REMOVE_ITEM DIR_SRCS "${TEST_SOURCE}")
    endif()
endforeach()
#编译可执行程序
ADD_EXECUTABLE(${PROJECT_NAME} ${DIR_SRCS})
target_include_directories(${PROJECT_NAME} PRIVATE "${SDK_ROOT}/include")
if(DLLLIB STREQUAL "lib" OR STATIC_LIB)
    target_compile_definitions(${PROJECT_NAME} PRIVATE STATIC_LIB)
endif()
if(LIBRARIESRELY)
    foreach(library IN LISTS LIBRARIESRELY)
        if(IS_ABSOLUTE "${library}")
            target_link_libraries(${PROJECT_NAME} PRIVATE "${library}")
        else()
            target_link_libraries(${PROJECT_NAME} PRIVATE "${SDK_ROOT}/lib/${library}")
        endif()
    endforeach()
else()
    foreach(component CTaskThreadManager Semaphore LockFreeQueue LockFreeHashMap LockFreeMap)
        platform_test_sdk_library(library ${component})
        target_link_libraries(${PROJECT_NAME} PRIVATE "${library}")
    endforeach()
endif()
platform_test_link(${PROJECT_NAME} "${SDK_ROOT}")
find_package(Threads REQUIRED)
target_link_libraries(${PROJECT_NAME} PRIVATE Threads::Threads)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_link_libraries(${PROJECT_NAME} PRIVATE rt)
    target_linux_arm_atomics(${PROJECT_NAME} PRIVATE)
endif()
