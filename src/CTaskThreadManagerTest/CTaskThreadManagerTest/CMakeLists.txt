#设置CMAKE最低版本
CMAKE_MINIMUM_REQUIRED(VERSION 3.1.1)
option(CTTM_WASM_SMOKE "Build the standalone Emscripten pthread smoke test" OFF)
option(CTTM_IN_PROCESS_SMOKE "Build the single-process test for a native or mobile SDK" OFF)
option(CTTM_IN_PROCESS_CASES "Build registered test cases for a single-process host" OFF)
if(CTTM_WASM_SMOKE OR CTTM_IN_PROCESS_SMOKE OR CTTM_IN_PROCESS_CASES)
    cmake_minimum_required(VERSION 3.15)
    project(CTaskWasmSmoke LANGUAGES CXX)
    if(CTTM_WASM_SMOKE AND NOT EMSCRIPTEN)
        message(FATAL_ERROR "CTTM_WASM_SMOKE requires Emscripten")
    endif()
    if(NOT CTTM_INCLUDE_ROOT OR NOT CTTM_LIBRARY_ROOT)
        message(FATAL_ERROR "Set CTTM_INCLUDE_ROOT and CTTM_LIBRARY_ROOT to the matched target SDK")
    endif()
    if(CTTM_IN_PROCESS_CASES)
        set(test_target CTaskThreadCases)
        file(GLOB case_sources CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/*.cpp")
        list(REMOVE_ITEM case_sources "${CMAKE_CURRENT_SOURCE_DIR}/CTaskWasmSmoke.cpp")
        add_executable(${test_target} ${case_sources})
        target_compile_definitions(${test_target} PRIVATE CTTM_TEST_SINGLE_PROCESS)
    else()
        set(test_target CTaskWasmSmoke)
        add_executable(${test_target} CTaskWasmSmoke.cpp CTaskWasmSmoke.h)
    endif()
    set_target_properties(${test_target} PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES)
    target_compile_definitions(${test_target} PRIVATE STATIC_LIB)
    target_include_directories(${test_target} PRIVATE "${CTTM_INCLUDE_ROOT}")
    foreach(component CTaskThreadManager Semaphore LockFreeQueue LockFreeHashMap LockFreeMap)
        unset(CTTM_${component}_ARCHIVE CACHE)
        find_file(CTTM_${component}_ARCHIVE NAMES ${component}.lib ${component}.a lib${component}.a PATHS "${CTTM_LIBRARY_ROOT}" NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH)
        if(CTTM_${component}_ARCHIVE)
            target_link_libraries(${test_target} PRIVATE "${CTTM_${component}_ARCHIVE}")
        elseif(component STREQUAL "CTaskThreadManager" OR component STREQUAL "Semaphore")
            message(FATAL_ERROR "Missing ${component} under CTTM_LIBRARY_ROOT")
        endif()
    endforeach()
    if(EMSCRIPTEN)
        set_target_properties(${test_target} PROPERTIES SUFFIX ".js")
        target_compile_options(${test_target} PRIVATE -pthread -fexceptions)
        if(CTTM_IN_PROCESS_CASES)
            target_link_options(${test_target} PRIVATE -sDEFAULT_PTHREAD_STACK_SIZE=1048576 -sSTACK_SIZE=1048576 -sSTACK_OVERFLOW_CHECK=2)
        endif()
        set(CTTM_WASM_WORKERS 4 CACHE STRING "Prestarted host workers, size for the chosen case")
        target_link_options(${test_target} PRIVATE -pthread -fexceptions -sPROXY_TO_PTHREAD -sPTHREAD_POOL_SIZE=${CTTM_WASM_WORKERS} -sENVIRONMENT=node -sEXIT_RUNTIME=1 -sINITIAL_MEMORY=67108864 -sALLOW_MEMORY_GROWTH=1)
    else()
        find_package(Threads REQUIRED)
        target_link_libraries(${test_target} PRIVATE Threads::Threads)
        if(WIN32)
            target_link_libraries(${test_target} PRIVATE psapi advapi32 user32 ole32 oleaut32 uuid)
        endif()
        if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
            set_target_properties(${test_target} PROPERTIES MACOSX_BUNDLE YES)
        endif()
    endif()
    return()
endif()

if("${BIT}" STREQUAL "native" OR CMAKE_TOOLCHAIN_FILE)
    MESSAGE(STATUS "using the architecture of the selected compiler/toolchain")
elseif("${BIT}" STREQUAL "32")
    MESSAGE(STATUS "using 32bits (x86)")
    SET(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -m32")
    SET(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -m32")
elseif("${BIT}" STREQUAL "64")
    MESSAGE(STATUS "using 64bits (x86_64)")
    SET(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -m64")
    SET(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -m64")
elseif("${BIT}" STREQUAL "arm32")
    MESSAGE(STATUS "using arm32 (armhf)")
    SET(CMAKE_C_COMPILER "arm-linux-gnueabihf-gcc")
    SET(CMAKE_CXX_COMPILER "arm-linux-gnueabihf-g++")
elseif("${BIT}" STREQUAL "arm64")
    MESSAGE(STATUS "using arm64 (aarch64)")
    SET(CMAKE_C_COMPILER "aarch64-linux-gnu-gcc")
    SET(CMAKE_CXX_COMPILER "aarch64-linux-gnu-g++")
else()
    MESSAGE(FATAL_ERROR "Unknown BIT value: ${BIT}. Please set BIT to 'native', '32', '64', 'arm32', or 'arm64'")
endif()
