function(add_net_peer_fixture_test name mode)
    if(EMSCRIPTEN)
        if(NOT LIBRARY_TEST_HOST OR NOT CMAKE_CROSSCOMPILING_EMULATOR)
            message(FATAL_ERROR "Wasm fixture tests require LIBRARY_TEST_HOST=ON and a Node CMAKE_CROSSCOMPILING_EMULATOR")
        endif()
        if(mode STREQUAL "--discovery-timers")
            set(host_mode "${mode}")
        endif()
        add_test(NAME ${name} COMMAND ${CMAKE_CROSSCOMPILING_EMULATOR}
            "${CMAKE_CURRENT_SOURCE_DIR}/NetPeerServiceHostTest.js" "$<TARGET_FILE:NetPeerTest>" ${host_mode} ${ARGN})
        set_tests_properties(${name} PROPERTIES TIMEOUT 90)
    else()
        add_platform_test(${name} NetPeerTest "${mode}" ${ARGN})
    endif()
endfunction()
