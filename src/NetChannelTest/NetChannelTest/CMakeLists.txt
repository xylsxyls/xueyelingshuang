cmake_minimum_required(VERSION 3.15)
if(BIT STREQUAL "wasm64")
    string(APPEND CMAKE_C_FLAGS " -sMEMORY64=1")
    string(APPEND CMAKE_CXX_FLAGS " -sMEMORY64=1")
    string(APPEND CMAKE_EXE_LINKER_FLAGS " -sMEMORY64=1")
endif()
include("${CMAKE_CURRENT_SOURCE_DIR}/../../PlatformTest/PlatformTest/src/PlatformTest.cmake")
# 原生测试和目标 SDK 必须使用同一 ABI，显式交叉工具链自行指定目标。
if(NOT CMAKE_TOOLCHAIN_FILE)
    if((NOT CMAKE_SYSTEM_NAME OR CMAKE_SYSTEM_NAME STREQUAL "Linux") AND CMAKE_HOST_SYSTEM_NAME STREQUAL "Linux")
        if(BIT MATCHES "^(32|64)$")
            string(APPEND CMAKE_CXX_FLAGS_INIT " -m${BIT}")
        else()
            platform_test_select_linux_arm()
        endif()
    elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
        platform_test_select_linux_arm()
    elseif(((NOT CMAKE_SYSTEM_NAME AND CMAKE_HOST_APPLE) OR CMAKE_SYSTEM_NAME MATCHES "^(Darwin|iOS)$")
        AND NOT CMAKE_OSX_ARCHITECTURES)
        if(BIT STREQUAL "arm64")
            set(CMAKE_OSX_ARCHITECTURES arm64 CACHE STRING "Apple SDK architecture")
        elseif(BIT STREQUAL "64")
            set(CMAKE_OSX_ARCHITECTURES x86_64 CACHE STRING "Apple SDK architecture")
        endif()
    endif()
endif()
project(NetChannelTest LANGUAGES CXX)
# BIT 也可随显式工具链传入；检查实际编译目标，避免同位宽异架构或复用缓存时测错 SDK。
if(BIT AND CMAKE_SYSTEM_NAME MATCHES "^(Darwin|iOS)$" AND NOT BIT MATCHES "^(64|arm64)$")
    message(FATAL_ERROR "Apple SDKs require BIT=64 or arm64")
endif()
# Linux ARM 的指针宽度、架构和硬浮点 ABI 由下方 build_sdk_test 检查。
if(BIT AND CMAKE_SYSTEM_NAME MATCHES "^(Linux|Darwin|iOS|Android)$"
    AND NOT (CMAKE_SYSTEM_NAME STREQUAL "Linux" AND BIT MATCHES "^arm(32|64)$"))
    function(check_net_channel_test_abi)
        if(BIT STREQUAL "32")
            set(arch "defined(__i386__)")
        elseif(BIT STREQUAL "64")
            set(arch "defined(__x86_64__)")
        elseif(BIT STREQUAL "arm32")
            set(arch "defined(__arm__)")
        elseif(BIT STREQUAL "arm64")
            set(arch "defined(__aarch64__) || defined(__arm64__)")
        else()
            message(FATAL_ERROR "Unsupported native BIT=${BIT}")
        endif()
        string(REGEX REPLACE "^arm" "" bits "${BIT}")
        math(EXPR pointer_bytes "${bits} / 8")
        set(CMAKE_TRY_COMPILE_TARGET_TYPE STATIC_LIBRARY)
        include(CheckCXXSourceCompiles)
        unset(NET_CHANNEL_TEST_ABI_MATCHES CACHE)
        check_cxx_source_compiles("#if !(${arch}) || __SIZEOF_POINTER__ != ${pointer_bytes}\n#error Wrong target ABI\n#endif\nint abiCheck() { return 0; }" NET_CHANNEL_TEST_ABI_MATCHES)
        if(NOT NET_CHANNEL_TEST_ABI_MATCHES)
            message(FATAL_ERROR "Compiler target disagrees with BIT=${BIT}. Use a build directory and SDK for that ABI.")
        endif()
    endfunction()
    check_net_channel_test_abi()
endif()
include(CTest)
build_sdk_test(NetChannelTest PROJECTS NetChannel)
