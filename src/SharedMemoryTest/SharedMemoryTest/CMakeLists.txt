cmake_minimum_required(VERSION 3.15)
project(SharedMemoryTest LANGUAGES CXX)
if(EMSCRIPTEN)
    message(FATAL_ERROR "This suite requires a native target")
endif()
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(root "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
include("${root}/src/PlatformTest/PlatformTest/src/PlatformTest.cmake")
if(NOT SDK_ROOT)
    set(SDK_ROOT "$ENV{CLOUD_REBUILD_SDK_ROOT}")
endif()
if(NOT SDK_ROOT)
    set(SDK_ROOT "${root}")
endif()
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:DebugDLL>")
endif()
add_executable(SharedMemoryTest SharedMemoryTest.cpp)
# 保留跨翻译单元的实际接口调用，静态 SDK 的热视图循环也需要测到每次调用。
set_property(TARGET SharedMemoryTest PROPERTY INTERPROCEDURAL_OPTIMIZATION FALSE)
platform_test_link(SharedMemoryTest "${SDK_ROOT}")
target_include_directories(SharedMemoryTest PRIVATE "${SDK_ROOT}/include")
platform_test_sdk_library(prepared_library SharedMemory)
target_link_libraries(SharedMemoryTest PRIVATE "${prepared_library}")
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_link_libraries(SharedMemoryTest PRIVATE rt)
    # 静态 SDK 的实际符号随 libc 和目标位宽而异；只包装归档中真实使用的调用。
    if(prepared_library MATCHES "\\.a$" AND CMAKE_NM)
        execute_process(COMMAND "${CMAKE_NM}" -u "${prepared_library}"
            RESULT_VARIABLE nm_status OUTPUT_VARIABLE imports)
        if(nm_status EQUAL 0 AND imports MATCHES "[ \t]U[ \t]+(posix_fallocate(64)?)[\r\n]")
            set(reserve_symbol "${CMAKE_MATCH_1}")
            if(imports MATCHES "[ \t]U[ \t]+(ftruncate(64)?)[\r\n]")
                set(resize_symbol "${CMAKE_MATCH_1}")
                target_link_options(SharedMemoryTest PRIVATE
                    "-Wl,--wrap=${reserve_symbol}" "-Wl,--wrap=${resize_symbol}")
                target_compile_definitions(SharedMemoryTest PRIVATE
                    SHM_REAL_RESERVE=__real_${reserve_symbol} SHM_WRAP_RESERVE=__wrap_${reserve_symbol}
                    SHM_REAL_RESIZE=__real_${resize_symbol} SHM_WRAP_RESIZE=__wrap_${resize_symbol})
            endif()
        endif()
    endif()
endif()
foreach(dependency IN LISTS LIBRARIESRELY)
    if(IS_ABSOLUTE "${dependency}")
        target_link_libraries(SharedMemoryTest PRIVATE "${dependency}")
    else()
        target_link_libraries(SharedMemoryTest PRIVATE "${SDK_ROOT}/lib/${dependency}")
    endif()
endforeach()
if(UNIX)
    find_package(Threads REQUIRED)
    target_link_libraries(SharedMemoryTest PRIVATE Threads::Threads)
endif()
enable_testing()
configure_platform_test(SharedMemoryTest SharedMemoryTest.cpp TAKES_ARGS)
add_platform_test(SharedMemoryTest SharedMemoryTest)
if(LIBRARY_TEST_BENCHMARKS)
    add_platform_test(SharedMemoryTestBenchmark SharedMemoryTest --benchmark)
    set_tests_properties(SharedMemoryTestBenchmark PROPERTIES LABELS performance RUN_SERIAL TRUE)
    add_custom_target(check_shared_memory_benchmarks
        COMMAND "${CMAKE_COMMAND}" "-DBUILD_DIR=${CMAKE_CURRENT_BINARY_DIR}" "-DCONFIG=$<CONFIG>"
            -P "${CMAKE_CURRENT_SOURCE_DIR}/CheckBenchmarks.cmake"
        DEPENDS SharedMemoryTest USES_TERMINAL VERBATIM)
endif()
if(WIN32)
    # CTest 在构建目录启动 EXE；让加载器使用本轮链接的 SDK 中的配套 DLL。
    string(REPLACE ";" "\\;" test_path "${SDK_ROOT}/lib;$ENV{PATH}")
    set_tests_properties(SharedMemoryTest PROPERTIES ENVIRONMENT "PATH=${test_path}")
    if(LIBRARY_TEST_BENCHMARKS)
        set_tests_properties(SharedMemoryTestBenchmark PROPERTIES ENVIRONMENT "PATH=${test_path}")
    endif()
endif()
