cmake_minimum_required(VERSION 3.15)
project(LockFreeHashMapTest LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# Windows 同位数子进程崩溃采集，不向生产头文件插入诊断代码。
if(WIN32)
    add_executable(LockFreeHashMapCrashRunner LockFreeHashMapCrashRunner.cpp)
    target_link_libraries(LockFreeHashMapCrashRunner PRIVATE dbghelp)
endif()

# 直接测试源头头文件，不读取或写入共享 SDK
set(LOCKFREEHASHMAP_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../../LockFreeHashMap/LockFreeHashMap/src" CACHE PATH "被测库头文件目录")
option(LOCKFREEHASHMAP_SINGLE_THREAD_TESTS "仅运行普通 Wasm 可执行的串行用例" OFF)
option(LOCKFREEHASHMAP_TEST_MODULE_SHARED "将用例装入测试专用动态模块" OFF)

if(NOT EMSCRIPTEN OR NOT LOCKFREEHASHMAP_SINGLE_THREAD_TESTS)
    find_package(Threads REQUIRED)
endif()

set(check_sources
    LockFreeHashMapTests.cpp
    LockFreeHashMapMigration.cpp
    LockFreeHashMapBoundaries.cpp
    LockFreeHashMapTestState.cpp
    LockFreeHashMapOptimizations.cpp)
list(APPEND check_sources LockFreeHashMapTestThread.cpp)
if(LOCKFREEHASHMAP_TEST_MODULE_SHARED)
    add_library(LockFreeHashMapChecks SHARED ${check_sources})
    target_compile_definitions(LockFreeHashMapChecks PUBLIC LOCKFREEHASHMAP_TEST_DLL PRIVATE LOCKFREEHASHMAP_TEST_EXPORT)
else()
    add_library(LockFreeHashMapChecks STATIC ${check_sources})
endif()
target_include_directories(LockFreeHashMapChecks PUBLIC "${LOCKFREEHASHMAP_SOURCE}")
target_compile_definitions(LockFreeHashMapChecks PUBLIC STATIC_LIB)
if(TARGET Threads::Threads)
    target_link_libraries(LockFreeHashMapChecks PUBLIC Threads::Threads)
endif()
if(LOCKFREEHASHMAP_SINGLE_THREAD_TESTS)
    target_compile_definitions(LockFreeHashMapChecks PUBLIC LOCKFREEHASHMAP_TEST_SINGLE_THREAD)
endif()
if(EMSCRIPTEN)
    target_compile_options(LockFreeHashMapChecks PUBLIC -fexceptions)
    target_link_options(LockFreeHashMapChecks PUBLIC -sDISABLE_EXCEPTION_CATCHING=0 -sEXIT_RUNTIME=1
        -sABORTING_MALLOC=0 -sSTACK_SIZE=1048576 -sINITIAL_MEMORY=134217728)
    if(LOCKFREEHASHMAP_SINGLE_THREAD_TESTS)
        target_link_options(LockFreeHashMapChecks PUBLIC -sALLOW_MEMORY_GROWTH=1 -sMAXIMUM_MEMORY=536870912)
    else()
        target_compile_options(LockFreeHashMapChecks PUBLIC -pthread)
        target_link_options(LockFreeHashMapChecks PUBLIC -pthread -sPTHREAD_POOL_SIZE=16 -sPROXY_TO_PTHREAD=1)
    endif()
endif()
add_executable(LockFreeHashMapTest LockFreeHashMapTest.cpp)
target_link_libraries(LockFreeHashMapTest PRIVATE LockFreeHashMapChecks)

# 控制器和性能程序独立编译，分配故障注入不进入性能计时进程
if(NOT EMSCRIPTEN)

    target_compile_definitions(LockFreeHashMapSingleThread PRIVATE STATIC_LIB)
    target_link_libraries(LockFreeHashMapSingleThread PRIVATE Threads::Threads)
    add_executable(LockFreeHashMapProgressTests LockFreeHashMapProgressTests.cpp LockFreeHashMapTestThread.cpp)
    target_include_directories(LockFreeHashMapProgressTests PRIVATE "${LOCKFREEHASHMAP_SOURCE}")
    target_compile_definitions(LockFreeHashMapProgressTests PRIVATE STATIC_LIB)
    target_link_libraries(LockFreeHashMapProgressTests PRIVATE Threads::Threads)
    add_executable(LockFreeHashMapRunner LockFreeHashMapRunner.cpp LockFreeHashMapTestThread.cpp)
    add_executable(LockFreeHashMapWorkloads LockFreeHashMapWorkloads.cpp LockFreeHashMapTestThread.cpp)
    add_executable(LockFreeHashMapBenchmark LockFreeHashMapBenchmark.cpp)

    add_executable(LockFreeHashMapReadCosts LockFreeHashMapReadCosts.cpp LockFreeHashMapTestThread.cpp)
    target_include_directories(LockFreeHashMapReadCosts PRIVATE "${LOCKFREEHASHMAP_SOURCE}")
    target_compile_definitions(LockFreeHashMapReadCosts PRIVATE STATIC_LIB)
    target_link_libraries(LockFreeHashMapReadCosts PRIVATE Threads::Threads)
    target_include_directories(LockFreeHashMapWorkloads PRIVATE "${LOCKFREEHASHMAP_SOURCE}")
    target_include_directories(LockFreeHashMapBenchmark PRIVATE "${LOCKFREEHASHMAP_SOURCE}")
    target_link_libraries(LockFreeHashMapRunner PRIVATE Threads::Threads)
    target_link_libraries(LockFreeHashMapWorkloads PRIVATE Threads::Threads)
    target_link_libraries(LockFreeHashMapThreadControl PRIVATE Threads::Threads)
endif()

enable_testing()
foreach(module_name Epoch Pool Rewrite)
    add_executable(LockFreeHashMap${module_name}Tests LockFreeHashMap${module_name}Tests.cpp LockFreeHashMapTestThread.cpp)
    if(module_name STREQUAL "Rewrite")
        target_sources(LockFreeHashMapRewriteTests PRIVATE LockFreeHashMapPoolTests.cpp)
        target_compile_definitions(LockFreeHashMapRewriteTests PRIVATE LOCKFREEHASHMAP_POOL_NO_MAIN)
    endif()
    target_include_directories(LockFreeHashMap${module_name}Tests PRIVATE "${LOCKFREEHASHMAP_SOURCE}")
    target_compile_definitions(LockFreeHashMap${module_name}Tests PRIVATE STATIC_LIB)
    if(module_name STREQUAL "Epoch" OR module_name STREQUAL "Rewrite")
        target_sources(LockFreeHashMap${module_name}Tests PRIVATE LockFreeHashMapTestState.cpp)
    endif()
    if(TARGET Threads::Threads)
        target_link_libraries(LockFreeHashMap${module_name}Tests PRIVATE Threads::Threads)
    endif()
    if(LOCKFREEHASHMAP_SINGLE_THREAD_TESTS)
        target_compile_definitions(LockFreeHashMap${module_name}Tests PRIVATE LOCKFREEHASHMAP_TEST_SINGLE_THREAD)
    endif()
    if(EMSCRIPTEN)
        target_compile_options(LockFreeHashMap${module_name}Tests PRIVATE -fexceptions)
        target_link_options(LockFreeHashMap${module_name}Tests PRIVATE -sDISABLE_EXCEPTION_CATCHING=0 -sEXIT_RUNTIME=1
            -sABORTING_MALLOC=0 -sSTACK_SIZE=1048576 -sINITIAL_MEMORY=134217728)
        if(NOT LOCKFREEHASHMAP_SINGLE_THREAD_TESTS)
            target_compile_options(LockFreeHashMap${module_name}Tests PRIVATE -pthread)
            target_link_options(LockFreeHashMap${module_name}Tests PRIVATE -pthread -sPTHREAD_POOL_SIZE=16 -sPROXY_TO_PTHREAD=1)
        endif()
    endif()
    add_test(NAME LockFreeHashMap_${module_name} COMMAND LockFreeHashMap${module_name}Tests)
    set_tests_properties(LockFreeHashMap_${module_name} PROPERTIES TIMEOUT 120)
endforeach()
foreach(test_id RANGE 1 10)
    if(LOCKFREEHASHMAP_SINGLE_THREAD_TESTS AND NOT test_id MATCHES "^(1|2|3|4|9)$")
        continue()
    endif()
    add_test(NAME LockFreeHashMap_${test_id} COMMAND LockFreeHashMapTest ${test_id})
    set_tests_properties(LockFreeHashMap_${test_id} PROPERTIES TIMEOUT 120)
endforeach()
