# 部分 32 位 ABI 的编译器把 64 位原子操作放在支持库中。交叉配置只编译和链接，不运行设备程序。
function(lockfreehashmap_check_atomic_link)
    include(CheckCXXSourceCompiles)
    set(CMAKE_TRY_COMPILE_TARGET_TYPE EXECUTABLE)
    set(atomic_probe "#include <atomic>
#include <cstdint>
std::atomic<std::uint64_t> counter(0);
int main() { return counter.is_lock_free() ? static_cast<int>(counter.fetch_add(1)) : 1; }")
    check_cxx_source_compiles("${atomic_probe}" LOCKFREEHASHMAP_NATIVE_ATOMICS_LINK)
    if(NOT LOCKFREEHASHMAP_NATIVE_ATOMICS_LINK)
        list(APPEND CMAKE_REQUIRED_LIBRARIES atomic)
        check_cxx_source_compiles("${atomic_probe}" LOCKFREEHASHMAP_LIBATOMIC_LINK)
        if(NOT LOCKFREEHASHMAP_LIBATOMIC_LINK)
            message(FATAL_ERROR "测试工具链无法链接 C++11 的 64 位原子操作")
        endif()
        link_libraries(atomic)
    endif()
endfunction()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux" OR ANDROID)
    lockfreehashmap_check_atomic_link()
endif()
