cmake_minimum_required(VERSION 3.15)
if(BIT STREQUAL "wasm64")
    string(APPEND CMAKE_C_FLAGS_INIT " -sMEMORY64=1")
    string(APPEND CMAKE_CXX_FLAGS_INIT " -sMEMORY64=1")
    string(APPEND CMAKE_EXE_LINKER_FLAGS_INIT " -sMEMORY64=1")
endif()
project(TimerTest LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(root "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
include("${root}/src/PlatformTest/PlatformTest/src/PlatformTest.cmake")
if(NOT SDK_ROOT)
    set(SDK_ROOT "$ENV{CLOUD_REBUILD_SDK_ROOT}")
endif()
if(NOT SDK_ROOT)
    set(SDK_ROOT "${root}")
endif()
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:DebugDLL>")
endif()
add_executable(TimerTest TimerTest.cpp)
platform_test_link(TimerTest "${SDK_ROOT}")
target_include_directories(TimerTest PRIVATE "${SDK_ROOT}/include")
if(EMSCRIPTEN)
    platform_test_wasm_headers(TimerTest "${SDK_ROOT}" Timer)
    math(EXPR wasm_bits "${CMAKE_SIZEOF_VOID_P} * 8")
    if(BIT AND NOT BIT STREQUAL "wasm${wasm_bits}")
        message(FATAL_ERROR "BIT disagrees with the compiler pointer size")
    endif()
    if(DEBUGRELEASE STREQUAL "debug" OR CMAKE_BUILD_TYPE STREQUAL "Debug")
        set(suffix d)
    endif()
    target_link_libraries(TimerTest PRIVATE "${SDK_ROOT}/lib/Timer${suffix}.a")
    target_compile_definitions(TimerTest PRIVATE STATIC_LIB)
    target_compile_options(TimerTest PRIVATE -fexceptions)
    target_link_options(TimerTest PRIVATE -fexceptions -sALLOW_MEMORY_GROWTH=1)
else()
    platform_test_sdk_library(prepared_library Timer)
    target_link_libraries(TimerTest PRIVATE "${prepared_library}")
    foreach(dependency IN LISTS LIBRARIESRELY)
        if(IS_ABSOLUTE "${dependency}")
            target_link_libraries(TimerTest PRIVATE "${dependency}")
        else()
            target_link_libraries(TimerTest PRIVATE "${SDK_ROOT}/lib/${dependency}")
        endif()
    endforeach()
    if(UNIX)
        find_package(Threads REQUIRED)
        target_link_libraries(TimerTest PRIVATE Threads::Threads)
    endif()
endif()
enable_testing()
add_library_platform_test(TimerTest TAKES_ARGS)
