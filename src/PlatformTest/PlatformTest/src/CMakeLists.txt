cmake_minimum_required(VERSION 3.15)
project(PlatformTestHostTools LANGUAGES CXX)
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
endif()
# 宿主与被测目标可能是不同架构。宿主单独准备 SDK，避免覆盖目标库。
set(host_sdk "${CMAKE_CURRENT_BINARY_DIR}/sdk")
configure_file(PlatformTestHost.hpp "${host_sdk}/include/PlatformTest/PlatformTestHost.hpp" COPYONLY)
add_library(PlatformTestHost STATIC PlatformTestHost.cpp)
target_compile_features(PlatformTestHost PUBLIC cxx_std_11)
target_include_directories(PlatformTestHost PUBLIC "${host_sdk}/include")
target_compile_definitions(PlatformTestHost PRIVATE PLATFORM_TEST_CMAKE="${CMAKE_COMMAND}")
add_executable(PlatformDeviceRunner PlatformDeviceRunner.cpp)
target_link_libraries(PlatformDeviceRunner PRIVATE PlatformTestHost)
if(PLATFORM_TEST_WITH_HTTP)
    get_filename_component(root "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
    set(http_source "${root}/src/HttpWorkTest/HttpWorkTest")
    add_executable(HttpWorkFixture "${http_source}/HttpWorkFixture.cpp" "${http_source}/HttpFixtureOrigin.cpp")
    target_link_libraries(HttpWorkFixture PRIVATE PlatformTestHost)
    target_compile_definitions(HttpWorkFixture PRIVATE PLATFORM_TEST_CMAKE="${CMAKE_COMMAND}")
    math(EXPR bits "${CMAKE_SIZEOF_VOID_P} * 8")
    set(cpu "${CMAKE_SYSTEM_PROCESSOR}")
    if(APPLE AND CMAKE_OSX_ARCHITECTURES)
        set(cpu "${CMAKE_OSX_ARCHITECTURES}")
    elseif(MSVC AND CMAKE_CXX_COMPILER_ARCHITECTURE_ID)
        set(cpu "${CMAKE_CXX_COMPILER_ARCHITECTURE_ID}")
    endif()
    string(TOLOWER "${cpu}" cpu)
    if(cpu MATCHES "^(i[3-6]86|x86|x64|amd64|x86_64)$")
        set(cpu_family intel)
    elseif(cpu MATCHES "^(arm|arm64|arm64e|aarch64|armv[5-8][a-z0-9_-]*)$")
        set(cpu_family arm)
    else()
        message(FATAL_ERROR "HttpWorkFixture requires one supported host CPU, got '${cpu}'")
    endif()

    if(WIN32 AND cpu_family STREQUAL "intel")
        set(platform windows)
        set(archive mbedtls.lib)
        target_link_libraries(HttpWorkFixture PRIVATE ws2_32 bcrypt)
    elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
        if(cpu_family STREQUAL "arm")
            set(platform arm)
        else()
            set(platform linux)
        endif()
        set(archive libmbedtls.a)
    elseif(CMAKE_SYSTEM_NAME STREQUAL "Darwin")
        if(cpu_family STREQUAL "arm")
            set(platform mac-apple-silicon)
        else()
            set(platform mac-intel)
        endif()
        set(archive libmbedtls.a)
    else()
        message(FATAL_ERROR "HttpWorkFixture has no host SDK for ${CMAKE_SYSTEM_NAME}/${cpu}")
    endif()
    if(WIN32)
        find_program(host_bash NAMES bash HINTS "$ENV{ProgramFiles}/Git/bin" "$ENV{ProgramW6432}/Git/bin" NO_DEFAULT_PATH)
    endif()
    find_program(host_bash NAMES bash)
    if(NOT host_bash)
        message(FATAL_ERROR "HttpWorkFixture requires Bash to run the mbedTLS before script")
    endif()
    execute_process(COMMAND "${CMAKE_COMMAND}" -E env
        "CLOUD_REBUILD_SDK_ROOT=${host_sdk}" "THIRDPARTY_PLATFORM=${platform}" THIRDPARTY_CRT=mt
        "${host_bash}" "${root}/src/CEncodeDecode/scripts/before_CEncodeDecode.sh" "${bits}" lib release only
        RESULT_VARIABLE prepare_result)
    if(NOT prepare_result STREQUAL "0")
        message(FATAL_ERROR "Cannot prepare ${platform} mbedTLS in ${host_sdk}")
    endif()
    target_include_directories(HttpWorkFixture PRIVATE "${host_sdk}/include/mbedtls")
    target_compile_definitions(HttpWorkFixture PRIVATE MBEDTLS_CONFIG_FILE="mbedtls_sdk_config.h")
    target_link_libraries(HttpWorkFixture PRIVATE "${host_sdk}/lib/${archive}")
    find_package(Threads REQUIRED)
    target_link_libraries(HttpWorkFixture PRIVATE Threads::Threads)
endif()
foreach(tool PlatformDeviceRunner HttpWorkFixture)
    if(TARGET ${tool})
        set_target_properties(${tool} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/$<0:>")
    endif()
endforeach()
