cmake_minimum_required(VERSION 3.15)
if(BIT STREQUAL "wasm64")
    string(APPEND CMAKE_C_FLAGS_INIT " -sMEMORY64=1")
    string(APPEND CMAKE_CXX_FLAGS_INIT " -sMEMORY64=1")
endif()
project(HttpWorkTest LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:DebugDLL>")
endif()
get_filename_component(root "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
include("${root}/src/PlatformTest/PlatformTest/src/PlatformTest.cmake")
if(NOT SDK_ROOT)
    set(SDK_ROOT "$ENV{CLOUD_REBUILD_SDK_ROOT}")
endif()
if(NOT SDK_ROOT)
    set(SDK_ROOT "${root}")
endif()
if(DEBUGRELEASE STREQUAL "debug" OR CMAKE_BUILD_TYPE STREQUAL "Debug")
    set(suffix d)
endif()
add_executable(HttpWorkTest HttpWorkTest.cpp HttpClientCases.cpp)
platform_test_link(HttpWorkTest "${SDK_ROOT}")
target_include_directories(HttpWorkTest PRIVATE "${SDK_ROOT}/include")
if(EMSCRIPTEN)
    platform_test_wasm_headers(HttpWorkTest "${SDK_ROOT}" HttpRequest CEncodeDecode)
    option(HTTPWORK_WASM_HOST "Build the modular browser/WeChat client test" OFF)
    if(HTTPWORK_WASM_HOST)
        target_link_options(HttpWorkTest PRIVATE -sMODULARIZE=1 -sEXPORT_NAME=createHttpWorkTest -sENVIRONMENT=web,node -sINVOKE_RUN=0
            "-sEXPORTED_RUNTIME_METHODS=[\"callMain\"]" "-sEXPORTED_FUNCTIONS=[\"_main\",\"_httpWorkTestResult\"]"
            --pre-js "${CMAKE_CURRENT_SOURCE_DIR}/HttpWorkWasmPre.js")
        configure_file(HttpWorkHost.js HttpWorkHost.js COPYONLY)
        foreach(file IN ITEMS LibraryTestWidth.js LibraryTestMemory64.wasm)
            configure_file("${root}/src/PlatformTest/PlatformTest/src/${file}" "${CMAKE_CURRENT_BINARY_DIR}/${file}" COPYONLY)
        endforeach()
    endif()
    math(EXPR wasm_bits "${CMAKE_SIZEOF_VOID_P} * 8")
    foreach(library IN ITEMS HttpRequest CEncodeDecode)
        target_link_libraries(HttpWorkTest PRIVATE "${SDK_ROOT}/lib/${library}${suffix}.a")
    endforeach()
    target_compile_definitions(HttpWorkTest PRIVATE STATIC_LIB HTTPWORK_CLIENT_ONLY)
    target_compile_options(HttpWorkTest PRIVATE -fexceptions)
    target_link_options(HttpWorkTest PRIVATE -fexceptions -sALLOW_MEMORY_GROWTH=1)
    target_link_libraries(HttpWorkTest PRIVATE "${SDK_ROOT}/lib/libmbedtls.a")
else()
    target_sources(HttpWorkTest PRIVATE HttpClientReuse.cpp)
    math(EXPR native_bits "${CMAKE_SIZEOF_VOID_P} * 8")
    if(APPLE)
        list(LENGTH CMAKE_OSX_ARCHITECTURES architecture_count)
        if(architecture_count GREATER 1)
            message(FATAL_ERROR "Prepare one HttpWorkTest SDK per Apple architecture")
        endif()
    endif()
    if(BIT)
        if(NOT BIT MATCHES "^(arm)?(32|64)$" OR NOT BIT MATCHES "${native_bits}$")
            message(FATAL_ERROR "BIT=${BIT} disagrees with the native target (${native_bits}-bit)")
        endif()
        set(native_tag "${BIT}")
    else()
        set(target_cpu "${CMAKE_SYSTEM_PROCESSOR}")
        if(CMAKE_ANDROID_ARCH_ABI)
            set(target_cpu "${CMAKE_ANDROID_ARCH_ABI}")
        elseif(CMAKE_OSX_ARCHITECTURES)
            set(target_cpu "${CMAKE_OSX_ARCHITECTURES}")
        elseif(CMAKE_GENERATOR_PLATFORM)
            set(target_cpu "${CMAKE_GENERATOR_PLATFORM}")
        endif()
        string(TOLOWER "${target_cpu}" target_cpu)
        if(target_cpu MATCHES "^(arm|aarch64)")
            set(native_tag "arm${native_bits}")
        elseif(target_cpu MATCHES "^(x86|x64|amd64|win32|i[3-6]86)")
            set(native_tag "${native_bits}")
        else()
            message(FATAL_ERROR "Set BIT=32, 64, arm32 or arm64 for ${target_cpu}")
        endif()
    endif()
    set(libraries HttpRequest)
    if(WIN32 OR CMAKE_SYSTEM_NAME STREQUAL "Linux")
        list(APPEND libraries HttpService CStringManager CSystem)
        target_sources(HttpWorkTest PRIVATE HttpServiceSocketBenchmark.cpp HttpServiceOptionsCases.cpp)
    else()
        target_compile_definitions(HttpWorkTest PRIVATE HTTPWORK_CLIENT_ONLY)
    endif()
    list(APPEND libraries CEncodeDecode)
    target_link_directories(HttpWorkTest PRIVATE "${SDK_ROOT}/lib")
    foreach(library IN LISTS libraries)
        if(MSVC)
            target_link_libraries(HttpWorkTest PRIVATE "${SDK_ROOT}/lib/${library}$<$<CONFIG:Debug>:d>.lib")
        else()
            platform_test_sdk_library(prepared_library "${library}")
            target_link_libraries(HttpWorkTest PRIVATE "${prepared_library}")
        endif()
    endforeach()
    foreach(dependency IN LISTS LIBRARIESRELY)
        if(IS_ABSOLUTE "${dependency}")
            target_link_libraries(HttpWorkTest PRIVATE "${dependency}")
        else()
            target_link_libraries(HttpWorkTest PRIVATE "${SDK_ROOT}/lib/${dependency}")
        endif()
    endforeach()
    if(WIN32)
        target_compile_definitions(HttpWorkTest PRIVATE STATIC_LIB)
        if(MSVC)
            set(suffix "$<$<CONFIG:Debug>:d>")
        endif()
        target_link_libraries(HttpWorkTest PRIVATE "civetweb${suffix}"
            "mbedtls${suffix}" "libcurl${suffix}"
            ws2_32 iphlpapi psapi userenv bcrypt shell32 ole32 oleaut32 uuid gdi32 user32 dbghelp)
    elseif(UNIX)
        target_link_libraries(HttpWorkTest PRIVATE "${SDK_ROOT}/lib/libmbedtls.a")
        if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
            target_link_libraries(HttpWorkTest PRIVATE "${SDK_ROOT}/lib/libcivetweb.a" ${CMAKE_DL_LIBS})
        elseif(APPLE)
            target_link_libraries(HttpWorkTest PRIVATE "-framework Security")
        endif()
        if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
            set(curl_archive "${SDK_ROOT}/lib/libcurl.so.4")
        elseif(APPLE OR ANDROID)
            set(curl_archive "${SDK_ROOT}/lib/libcurl.a")
        else()
            message(FATAL_ERROR "Choose Linux, Android, macOS or iOS for the native Unix HttpWorkTest")
        endif()
        if(NOT EXISTS "${curl_archive}" OR IS_DIRECTORY "${curl_archive}")
            message(FATAL_ERROR "Missing prepared native library: ${curl_archive}. Run before/version for this target first.")
        endif()
        target_link_libraries(HttpWorkTest PRIVATE "${curl_archive}")
        if(APPLE OR ANDROID)
            # 静态 curl 的 TLS 后端及系统框架取自该归档对应的目标包。
            set(curl_package_dir "${SDK_ROOT}/lib/cmake/CURL")
            if(CURL_DIR)
                set(curl_package_dir "${CURL_DIR}")
            endif()
            if(NOT EXISTS "${curl_package_dir}/CURLConfig.cmake" AND NOT EXISTS "${curl_package_dir}/curl-config.cmake")
                message(FATAL_ERROR "Missing CURL 8.23.0 target package in ${curl_package_dir}. Set CURL_DIR to its CMake package directory.")
            endif()
            set(curl_explicit_dir "${CURL_DIR}")
            find_package(CURL 8.23.0 EXACT CONFIG REQUIRED PATHS "${curl_package_dir}" NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH)
            if(NOT curl_explicit_dir)
                unset(CURL_DIR CACHE)
            endif()
            if(NOT TARGET CURL::libcurl_static)
                message(FATAL_ERROR "${curl_package_dir} must export CURL::libcurl_static")
            endif()
            get_target_property(curl_package_archive CURL::libcurl_static LOCATION_RELEASE)
            if(NOT EXISTS "${curl_package_archive}" OR IS_DIRECTORY "${curl_package_archive}")
                message(FATAL_ERROR "${curl_package_dir} has no release archive for CURL::libcurl_static")
            endif()
            file(SHA256 "${curl_archive}" staged_hash)
            file(SHA256 "${curl_package_archive}" package_hash)
            if(NOT package_hash STREQUAL staged_hash)
                message(FATAL_ERROR "${curl_package_dir} does not describe the staged archive ${curl_archive}")
            endif()
            get_target_property(curl_dependencies CURL::libcurl_static INTERFACE_LINK_LIBRARIES)
            get_target_property(curl_options CURL::libcurl_static INTERFACE_LINK_OPTIONS)
            get_target_property(curl_directories CURL::libcurl_static INTERFACE_LINK_DIRECTORIES)
            if(curl_dependencies)
                target_link_libraries(HttpWorkTest PRIVATE ${curl_dependencies})
            endif()
            if(curl_options)
                target_link_options(HttpWorkTest PRIVATE ${curl_options})
            endif()
            if(curl_directories)
                target_link_directories(HttpWorkTest PRIVATE ${curl_directories})
            endif()
        endif()
        find_package(Threads REQUIRED)
        target_link_libraries(HttpWorkTest PRIVATE Threads::Threads)
    endif()
endif()
include(CTest)
if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
    configure_platform_test(HttpWorkTest HttpWorkTest.cpp TAKES_ARGS LOCAL_NETWORK)
endif()
if(BUILD_TESTING)
    option(LIBRARY_TEST_BENCHMARKS "Register performance cases with CTest" OFF)
    platform_test_host_tools(HttpWorkTest ON)
    get_property(http_fixture GLOBAL PROPERTY HTTPWORK_FIXTURE)
    get_property(device_runner GLOBAL PROPERTY PLATFORM_DEVICE_RUNNER)
    if(WIN32 OR CMAKE_SYSTEM_NAME STREQUAL "Linux")
        add_test(NAME HttpServiceLoopback COMMAND HttpWorkTest)
        if(LIBRARY_TEST_BENCHMARKS)
            add_test(NAME HttpServiceBenchmark COMMAND HttpWorkTest --benchmark)
            set_tests_properties(HttpServiceBenchmark PROPERTIES LABELS performance)
        endif()
    endif()
    if(ANDROID OR CMAKE_SYSTEM_NAME STREQUAL "iOS")
        set(HTTPWORK_TEST_ORIGIN "" CACHE STRING "Fixture origin reachable from the target device")
        if(HTTPWORK_TEST_ORIGIN)
            string(REGEX REPLACE "/+$" "" fixture_origin "${HTTPWORK_TEST_ORIGIN}")
            add_platform_test(HttpRequestLoopback HttpWorkTest --client "${fixture_origin}")
            if(LIBRARY_TEST_BENCHMARKS)
                add_platform_test(HttpRequestBenchmark HttpWorkTest --benchmark "${fixture_origin}")
            endif()
        else()
            set(PLATFORM_TEST_DEVICE "" CACHE STRING "Android serial, iOS device UDID or simulator UDID")
            set(device_arguments)
            if(PLATFORM_TEST_DEVICE)
                list(APPEND device_arguments --device "${PLATFORM_TEST_DEVICE}")
            endif()
            if(ANDROID)
                set(device_mode android)
                set(device_product "$<TARGET_FILE:HttpWorkTest>")
                list(APPEND device_arguments --sdk-lib "${SDK_ROOT}/lib")
                set(PLATFORM_TEST_RUNTIME_LIBRARIES "" CACHE STRING "Additional Android test runtime libraries")
                foreach(runtime IN LISTS PLATFORM_TEST_RUNTIME_LIBRARIES)
                    list(APPEND device_arguments --runtime "${runtime}")
                endforeach()
            else()
                if(CMAKE_OSX_SYSROOT MATCHES "[Ss]imulator")
                    set(device_mode ios-simulator)
                else()
                    set(device_mode ios-device)
                endif()
                set(device_product "$<TARGET_BUNDLE_DIR:HttpWorkTest>")
                get_target_property(device_identifier HttpWorkTest MACOSX_BUNDLE_GUI_IDENTIFIER)
                list(APPEND device_arguments --identifier "${device_identifier}")
            endif()
            add_test(NAME HttpRequestLoopback COMMAND "${http_fixture}"
                device "${device_runner}" ${device_mode} "${device_product}" ${device_arguments})
            set_tests_properties(HttpRequestLoopback PROPERTIES TIMEOUT 240 RESOURCE_LOCK platform_device_HttpWorkTest)
            if(LIBRARY_TEST_BENCHMARKS)
                add_test(NAME HttpRequestBenchmark COMMAND "${http_fixture}"
                    device-benchmark "${device_runner}" ${device_mode} "${device_product}" ${device_arguments})
                set_tests_properties(HttpRequestBenchmark PROPERTIES TIMEOUT 240 RESOURCE_LOCK platform_device_HttpWorkTest)
            endif()
        endif()
    elseif(NOT CMAKE_CROSSCOMPILING OR EMSCRIPTEN)
        set(fixture_options)
        find_program(test_openssl openssl)
        if(test_openssl)
            list(APPEND fixture_options --openssl "${test_openssl}")
        endif()
        if(EMSCRIPTEN)
            list(APPEND fixture_options --node-ca)
        endif()
        set(fixture_driver)
        if(EMSCRIPTEN AND HTTPWORK_WASM_HOST)
            set(fixture_driver "${CMAKE_CURRENT_BINARY_DIR}/HttpWorkHost.js")
        endif()
        add_test(NAME HttpRequestLoopback COMMAND "${http_fixture}"
            ${fixture_options} ${CMAKE_CROSSCOMPILING_EMULATOR} ${fixture_driver} "$<TARGET_FILE:HttpWorkTest>")
        if(LIBRARY_TEST_BENCHMARKS)
            add_test(NAME HttpRequestBenchmark COMMAND "${http_fixture}" --benchmark
                ${CMAKE_CROSSCOMPILING_EMULATOR} ${fixture_driver} "$<TARGET_FILE:HttpWorkTest>")
        endif()
    endif()
    if(TEST HttpRequestBenchmark)
        set_tests_properties(HttpRequestBenchmark PROPERTIES LABELS performance)
    endif()
endif()
