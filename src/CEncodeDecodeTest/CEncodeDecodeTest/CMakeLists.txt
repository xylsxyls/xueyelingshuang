cmake_minimum_required(VERSION 3.15)
if(BIT STREQUAL "wasm64")
    string(APPEND CMAKE_C_FLAGS_INIT " -sMEMORY64=1")
    string(APPEND CMAKE_CXX_FLAGS_INIT " -sMEMORY64=1")
    string(APPEND CMAKE_EXE_LINKER_FLAGS_INIT " -sMEMORY64=1")
endif()
project(CEncodeDecodeTest LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(root "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
include("${root}/src/PlatformTest/PlatformTest/src/PlatformTest.cmake")
if(NOT SDK_ROOT)
    set(SDK_ROOT "$ENV{CLOUD_REBUILD_SDK_ROOT}")
endif()
if(NOT SDK_ROOT)
    set(SDK_ROOT "${root}")
endif()
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:DebugDLL>")
endif()
add_executable(CEncodeDecodeTest CEncodeDecodeTest.cpp
    CEncodeDecodeHashCases.cpp CEncodeDecodeCertificateCases.cpp)
platform_test_link(CEncodeDecodeTest "${SDK_ROOT}")
target_include_directories(CEncodeDecodeTest PRIVATE "${SDK_ROOT}/include" "${SDK_ROOT}/include/mbedtls"
    "${CMAKE_CURRENT_SOURCE_DIR}")
if(EMSCRIPTEN)
    platform_test_wasm_headers(CEncodeDecodeTest "${SDK_ROOT}" CEncodeDecode)
    math(EXPR wasm_bits "${CMAKE_SIZEOF_VOID_P} * 8")
    if(BIT AND NOT BIT STREQUAL "wasm${wasm_bits}")
        message(FATAL_ERROR "BIT disagrees with the compiler pointer size")
    endif()
    if(DEBUGRELEASE STREQUAL "debug" OR CMAKE_BUILD_TYPE STREQUAL "Debug")
        set(suffix d)
    endif()
    target_link_libraries(CEncodeDecodeTest PRIVATE "${SDK_ROOT}/lib/CEncodeDecode${suffix}.a")
    target_compile_definitions(CEncodeDecodeTest PRIVATE STATIC_LIB)
    target_compile_options(CEncodeDecodeTest PRIVATE -fexceptions)
    target_link_options(CEncodeDecodeTest PRIVATE -fexceptions -sALLOW_MEMORY_GROWTH=1)
    target_link_libraries(CEncodeDecodeTest PRIVATE "${SDK_ROOT}/lib/libmbedtls.a")
else()
    if(MSVC)
        target_link_directories(CEncodeDecodeTest PRIVATE "${SDK_ROOT}/lib")
        target_link_libraries(CEncodeDecodeTest PRIVATE "${SDK_ROOT}/lib/CEncodeDecode$<$<CONFIG:Debug>:d>.lib"
            "${SDK_ROOT}/lib/mbedtls$<$<CONFIG:Debug>:d>.lib")
    else()
        platform_test_sdk_library(prepared_library CEncodeDecode)
        target_link_libraries(CEncodeDecodeTest PRIVATE "${prepared_library}")
    endif()
    if(WIN32)
        target_link_libraries(CEncodeDecodeTest PRIVATE bcrypt)
    elseif(UNIX)
        target_link_libraries(CEncodeDecodeTest PRIVATE "${SDK_ROOT}/lib/libmbedtls.a")
    endif()
    foreach(dependency IN LISTS LIBRARIESRELY)
        if(IS_ABSOLUTE "${dependency}")
            target_link_libraries(CEncodeDecodeTest PRIVATE "${dependency}")
        else()
            target_link_libraries(CEncodeDecodeTest PRIVATE "${SDK_ROOT}/lib/${dependency}")
        endif()
    endforeach()
endif()
option(CENCODE_TEST_PRIVATE_CASES "Run private algorithm cases with a desktop static CEncodeDecode SDK" OFF)
if(CENCODE_TEST_PRIVATE_CASES)
    if(EMSCRIPTEN OR ANDROID OR CMAKE_SYSTEM_NAME STREQUAL "iOS")
        message(FATAL_ERROR "Private cases require a desktop static CEncodeDecode SDK")
    endif()
    target_sources(CEncodeDecodeTest PRIVATE CEncodeDecodeCompatibilityCases.cpp)
    target_compile_definitions(CEncodeDecodeTest PRIVATE CENCODE_TEST_PRIVATE_CASES)
endif()
enable_testing()
add_library_platform_test(CEncodeDecodeTest TAKES_ARGS)
