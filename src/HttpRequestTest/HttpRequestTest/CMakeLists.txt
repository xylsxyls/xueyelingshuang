cmake_minimum_required(VERSION 3.15)
if(BIT STREQUAL "wasm64")
    string(APPEND CMAKE_C_FLAGS " -sMEMORY64=1")
    string(APPEND CMAKE_CXX_FLAGS " -sMEMORY64=1")
    string(APPEND CMAKE_EXE_LINKER_FLAGS " -sMEMORY64=1")
endif()
include("${CMAKE_CURRENT_SOURCE_DIR}/../../PlatformTest/PlatformTest/src/PlatformTest.cmake")
platform_test_select_linux_arm()
project(HttpRequestTest LANGUAGES CXX)
include(CTest)
if(MSVC)
    set(vendors libcurl$<$<CONFIG:Debug>:d>.lib mbedtls$<$<CONFIG:Debug>:d>.lib)
    set(system_libraries ws2_32 iphlpapi psapi userenv bcrypt shell32 ole32 oleaut32 uuid gdi32 user32 dbghelp advapi32 crypt32 secur32 normaliz wldap32)
else()
    set(vendors libmbedtls.a)
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
        list(APPEND vendors libcurl.so.4)
    elseif(APPLE OR ANDROID)
        find_package(CURL CONFIG REQUIRED)
        list(APPEND system_libraries CURL::libcurl_static)
    endif()
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
        list(APPEND system_libraries rt)
    elseif(APPLE)
        list(APPEND system_libraries "-framework Security")
    endif()
endif()
build_sdk_test(HttpRequestTest PROJECTS HttpRequest CEncodeDecode
    ARCHIVES ${vendors} SYSTEM_LIBRARIES ${system_libraries})
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    # 在目标机运行时使用构建选定的 curl 及其 TLS 后端。
    if(NOT SDK_ROOT)
        set(SDK_ROOT "$ENV{CLOUD_REBUILD_SDK_ROOT}")
    endif()
    if(NOT SDK_ROOT)
        get_filename_component(SDK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
    endif()
    set_target_properties(HttpRequestTest PROPERTIES BUILD_WITH_INSTALL_RPATH ON INSTALL_RPATH "$ORIGIN")
    add_custom_command(TARGET HttpRequestTest POST_BUILD
        COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${SDK_ROOT}/lib/libcurl.so.4"
            "$<TARGET_FILE_DIR:HttpRequestTest>/libcurl.so.4" VERBATIM)
endif()
