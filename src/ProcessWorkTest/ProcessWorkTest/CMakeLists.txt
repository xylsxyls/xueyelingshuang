cmake_minimum_required(VERSION 3.15)
project(ProcessWorkTest LANGUAGES CXX)
if(EMSCRIPTEN)
    message(FATAL_ERROR "ProcessWorkTest requires native shared memory and processes")
endif()
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(root "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
include("${root}/src/PlatformTest/PlatformTest/src/PlatformTest.cmake")
if(NOT SDK_ROOT)
    set(SDK_ROOT "$ENV{CLOUD_REBUILD_SDK_ROOT}")
endif()
if(NOT SDK_ROOT)
    set(SDK_ROOT "${root}")
endif()
if(NOT CMAKE_CONFIGURATION_TYPES AND NOT CMAKE_BUILD_TYPE)
    if(DEBUGRELEASE STREQUAL "debug")
        set(CMAKE_BUILD_TYPE Debug)
    else()
        set(CMAKE_BUILD_TYPE Release)
    endif()
endif()
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:DebugDLL>")
endif()
set(support "${root}/src/CTaskThreadManagerTest/CTaskThreadManagerTest")
file(GLOB suite CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/*.cpp")
add_executable(ProcessWorkTest ${suite} "${support}/BlackBoxTestSupport.cpp")
target_compile_definitions(ProcessWorkTest PRIVATE PROCESS_WORK_BENCH_IN_SUITE)
set(targets ProcessWorkTest)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
    add_executable(ProcessWorkBenchmark ProcessWorkBenchmark.cpp "${support}/BlackBoxTestSupport.cpp")
    list(APPEND targets ProcessWorkBenchmark)
endif()
find_package(Threads REQUIRED)
foreach(target IN LISTS targets)
    target_include_directories(${target} PRIVATE "${SDK_ROOT}/include" "${support}")
    if(DLLLIB STREQUAL "lib" OR STATIC_LIB)
        target_compile_definitions(${target} PRIVATE STATIC_LIB)
    endif()
    foreach(component IN ITEMS ProcessWork SharedMemory CTaskThreadManager CorrespondParam CStringManager CSystem LockFreeMap LockFreeHashMap LockFreeQueue Semaphore)
        platform_test_sdk_library(library ${component})
        target_link_libraries(${target} PRIVATE "${library}")
    endforeach()
    target_link_libraries(${target} PRIVATE Threads::Threads)
    foreach(dependency IN LISTS LIBRARIESRELY)
        if(IS_ABSOLUTE "${dependency}")
            target_link_libraries(${target} PRIVATE "${dependency}")
        else()
            target_link_libraries(${target} PRIVATE "${SDK_ROOT}/lib/${dependency}")
        endif()
    endforeach()
    if(WIN32)
        target_compile_definitions(${target} PRIVATE NOMINMAX)
        target_link_libraries(${target} PRIVATE psapi advapi32 user32 ole32 oleaut32 uuid)
    elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
        target_link_libraries(${target} PRIVATE rt)
    endif()
    if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
        target_compile_definitions(${target} PRIVATE CTTM_TEST_SINGLE_PROCESS)
    endif()
endforeach()
platform_test_link(ProcessWorkTest "${SDK_ROOT}")
enable_testing()
configure_platform_test(ProcessWorkTest ProcessWorkTest.cpp TAKES_ARGS)
if(ANDROID OR CMAKE_SYSTEM_NAME STREQUAL "iOS")
    # 设备宿主按次限时。逐用例启动，避免把整套回归计入单次设备超时。
    foreach(id RANGE 1 110)
        add_platform_test(ProcessWorkCase${id} ProcessWorkTest --case ${id})
        set_tests_properties(ProcessWorkCase${id} PROPERTIES SKIP_RETURN_CODE 77)
    endforeach()
else()
    add_platform_test(ProcessWorkTest ProcessWorkTest --all)
    set_tests_properties(ProcessWorkTest PROPERTIES TIMEOUT 1200)
endif()
