cmake_minimum_required(VERSION 3.15)

# Memory64需要在编译器探测前选择指针ABI。
if(BIT STREQUAL "wasm64")
    string(APPEND CMAKE_C_FLAGS_INIT " -sMEMORY64=1")
    string(APPEND CMAKE_CXX_FLAGS_INIT " -sMEMORY64=1")
    string(APPEND CMAKE_EXE_LINKER_FLAGS_INIT " -sMEMORY64=1")
endif()
project(MxmlCppTest LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(root "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
include("${root}/src/PlatformTest/PlatformTest/src/PlatformTest.cmake")
if(NOT SDK_ROOT)
    set(SDK_ROOT "$ENV{CLOUD_REBUILD_SDK_ROOT}")
endif()
if(NOT SDK_ROOT)
    set(SDK_ROOT "${root}")
endif()

file(TO_CMAKE_PATH "${SDK_ROOT}" SDK_ROOT)
add_executable(MxmlCppTest MxmlCppTest.cpp)
platform_test_link(MxmlCppTest "${SDK_ROOT}")
target_include_directories(MxmlCppTest PRIVATE "${SDK_ROOT}/include")
if(EMSCRIPTEN)
    platform_test_wasm_headers(MxmlCppTest "${SDK_ROOT}" MxmlCpp)
    math(EXPR wasm_bits "${CMAKE_SIZEOF_VOID_P} * 8")
    if(BIT AND NOT BIT STREQUAL "wasm${wasm_bits}")
        message(FATAL_ERROR "BIT and the Emscripten compiler pointer size disagree")
    endif()
    set(suffix "$<$<CONFIG:Debug>:d>")
    if(NOT CMAKE_CONFIGURATION_TYPES AND NOT CMAKE_BUILD_TYPE AND DEBUGRELEASE STREQUAL "debug")
        set(suffix d)
    endif()
    target_compile_definitions(MxmlCppTest PRIVATE STATIC_LIB)
    target_compile_options(MxmlCppTest PRIVATE -fexceptions)
    target_link_options(MxmlCppTest PRIVATE -fexceptions -sALLOW_MEMORY_GROWTH=1)
    target_link_libraries(MxmlCppTest PRIVATE "${SDK_ROOT}/lib/MxmlCpp${suffix}.a" "${SDK_ROOT}/lib/libmxml.a")
else()
    platform_test_sdk_library(prepared_library MxmlCpp)
    target_link_libraries(MxmlCppTest PRIVATE "${prepared_library}")
    if(MSVC)
        set_property(TARGET MxmlCppTest PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:DebugDLL>")
        target_link_directories(MxmlCppTest PRIVATE "${SDK_ROOT}/lib")
        target_link_libraries(MxmlCppTest PRIVATE "${SDK_ROOT}/lib/mxmlstat$<$<CONFIG:Debug>:d>.lib")
    else()
        target_link_libraries(MxmlCppTest PRIVATE "${SDK_ROOT}/lib/libmxml.a")
        if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
            find_package(Threads REQUIRED)
            target_link_libraries(MxmlCppTest PRIVATE Threads::Threads)
        endif()
    endif()
endif()
include(CTest)
if(BUILD_TESTING)
    add_library_platform_test(MxmlCppTest TAKES_ARGS)
endif()
