cmake_minimum_required(VERSION 3.15)
if(BIT STREQUAL "wasm64")
    string(APPEND CMAKE_C_FLAGS_INIT " -sMEMORY64=1")
    string(APPEND CMAKE_CXX_FLAGS_INIT " -sMEMORY64=1")
    string(APPEND CMAKE_EXE_LINKER_FLAGS_INIT " -sMEMORY64=1")
endif()
project(ProtoMessageManagerTest LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(root "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
include("${root}/src/PlatformTest/PlatformTest/src/PlatformTest.cmake")
if(NOT SDK_ROOT)
    set(SDK_ROOT "$ENV{CLOUD_REBUILD_SDK_ROOT}")
endif()
if(NOT SDK_ROOT)
    set(SDK_ROOT "${root}")
endif()
get_filename_component(SDK_ROOT "${SDK_ROOT}" ABSOLUTE)
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:DebugDLL>")
endif()
add_executable(ProtoMessageManagerTest ProtoMessageManagerTest.cpp)
platform_test_link(ProtoMessageManagerTest "${SDK_ROOT}")
target_include_directories(ProtoMessageManagerTest PRIVATE "${SDK_ROOT}/include/protobuf" "${SDK_ROOT}/include")
if(EMSCRIPTEN)
    platform_test_wasm_headers(ProtoMessageManagerTest "${SDK_ROOT}" ProtoMessageManager Variant)
    math(EXPR wasm_bits "${CMAKE_SIZEOF_VOID_P} * 8")
    if(BIT AND NOT BIT STREQUAL "wasm${wasm_bits}")
        message(FATAL_ERROR "BIT disagrees with the compiler pointer size")
    endif()
    if(DEBUGRELEASE STREQUAL "debug" OR CMAKE_BUILD_TYPE STREQUAL "Debug")
        set(suffix d)
    endif()
    foreach(library IN ITEMS ProtoMessageManager Variant)
        target_link_libraries(ProtoMessageManagerTest PRIVATE "${SDK_ROOT}/lib/${library}${suffix}.a")
    endforeach()
    target_compile_definitions(ProtoMessageManagerTest PRIVATE STATIC_LIB)
    target_compile_options(ProtoMessageManagerTest PRIVATE -fexceptions)
    target_link_options(ProtoMessageManagerTest PRIVATE -fexceptions -sALLOW_MEMORY_GROWTH=1)
    target_link_libraries(ProtoMessageManagerTest PRIVATE "${SDK_ROOT}/lib/libprotobuf-lite.a")
else()
    platform_test_sdk_library(prepared_library ProtoMessageManager)
    if(MSVC)
        target_link_directories(ProtoMessageManagerTest PRIVATE "${SDK_ROOT}/lib")
    endif()
    target_link_libraries(ProtoMessageManagerTest PRIVATE "${prepared_library}")
    if(UNIX AND NOT LIBRARIESRELY)
        platform_test_sdk_library(prepared_variant Variant)
        set(LIBRARIESRELY "${prepared_variant}" libprotobuf-lite.a)
    endif()
    foreach(dependency IN LISTS LIBRARIESRELY)
        if(IS_ABSOLUTE "${dependency}")
            target_link_libraries(ProtoMessageManagerTest PRIVATE "${dependency}")
        else()
            target_link_libraries(ProtoMessageManagerTest PRIVATE "${SDK_ROOT}/lib/${dependency}")
        endif()
    endforeach()
    if(UNIX)
        find_package(Threads REQUIRED)
        target_link_libraries(ProtoMessageManagerTest PRIVATE Threads::Threads)
    endif()
endif()
enable_testing()
add_library_platform_test(ProtoMessageManagerTest TAKES_ARGS)
