cmake_minimum_required(VERSION 3.15)
if(BIT STREQUAL "wasm64")
    string(APPEND CMAKE_C_FLAGS_INIT " -sMEMORY64=1")
    string(APPEND CMAKE_CXX_FLAGS_INIT " -sMEMORY64=1")
    string(APPEND CMAKE_EXE_LINKER_FLAGS_INIT " -sMEMORY64=1")
endif()
project(CStringManagerTest LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(root "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
if(NOT SDK_ROOT)
    set(SDK_ROOT "$ENV{CLOUD_REBUILD_SDK_ROOT}")
endif()
if(NOT SDK_ROOT)
    set(SDK_ROOT "${root}")
endif()
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:DebugDLL>")
endif()
add_executable(CStringManagerTest CStringManagerTest.cpp CStringManagerRegressionTests.cpp)
target_include_directories(CStringManagerTest PRIVATE "${SDK_ROOT}/include")
if(EMSCRIPTEN)
    math(EXPR wasm_bits "${CMAKE_SIZEOF_VOID_P} * 8")
    if(BIT AND NOT BIT STREQUAL "wasm${wasm_bits}")
        message(FATAL_ERROR "BIT disagrees with the compiler pointer size")
    endif()
    if(DEBUGRELEASE STREQUAL "debug" OR CMAKE_BUILD_TYPE STREQUAL "Debug")
        set(suffix d)
    endif()
    # before/version准备目标归档。Wasm SDK仅发布元数据，调用方从同版工程取得C++头文件。
    foreach(library IN ITEMS CStringManager)
        file(GLOB headers CONFIGURE_DEPENDS "${root}/src/${library}/${library}/src/*.h" "${root}/src/${library}/${library}/src/*.inl")
        foreach(header IN LISTS headers)
            get_filename_component(filename "${header}" NAME)
            configure_file("${header}" "${CMAKE_CURRENT_BINARY_DIR}/headers/${library}/${filename}" COPYONLY)
        endforeach()
        target_link_libraries(CStringManagerTest PRIVATE "${SDK_ROOT}/lib/${library}${suffix}.a")
    endforeach()
    target_include_directories(CStringManagerTest BEFORE PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/headers")
    target_compile_definitions(CStringManagerTest PRIVATE STATIC_LIB)
    target_compile_options(CStringManagerTest PRIVATE -fexceptions)
    target_link_options(CStringManagerTest PRIVATE -fexceptions -sALLOW_MEMORY_GROWTH=1)
else()
    include("${root}/common/PlatformTest/PreparedLibrary.cmake")
    platform_test_sdk_library(prepared_library CStringManager)
    target_link_libraries(CStringManagerTest PRIVATE "${prepared_library}")
    if(MSVC)
        target_link_directories(CStringManagerTest PRIVATE "${SDK_ROOT}/lib")
    endif()
    foreach(dependency IN LISTS LIBRARIESRELY)
        if(IS_ABSOLUTE "${dependency}")
            target_link_libraries(CStringManagerTest PRIVATE "${dependency}")
        else()
            target_link_libraries(CStringManagerTest PRIVATE "${SDK_ROOT}/lib/${dependency}")
        endif()
    endforeach()
    if(UNIX)
        find_package(Threads REQUIRED)
        target_link_libraries(CStringManagerTest PRIVATE Threads::Threads)
    endif()
endif()
enable_testing()
include("${root}/common/wasm/LibraryTestHost.cmake")
add_library_platform_test(CStringManagerTest TAKES_ARGS)
