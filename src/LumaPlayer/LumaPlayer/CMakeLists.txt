#设置CMAKE最低版本
CMAKE_MINIMUM_REQUIRED(VERSION 3.10.1)
SET(CMAKE_CXX_STANDARD 11)

SET(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -fPIC")
SET(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -fPIC")

#SET(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -finput-charset='gb2312'")
#SET(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -finput-charset='gb2312'")

if(${DEBUGRELEASE} STREQUAL "debug")
#设置项目名称
SET(PROJECT_NAME LumaPlayerd)
SET(CMAKE_CXX_FLAGS_DEBUG "$ENV{CXXFLAGS} -Wl,-rpath=. -O0 -g -ggdb ")
SET(CMAKE_BUILD_TYPE "Debug")
elseif(${DEBUGRELEASE} STREQUAL "release")
#设置项目名称
SET(PROJECT_NAME LumaPlayer)
SET(CMAKE_CXX_FLAGS_RELEASE "$ENV{CXXFLAGS} -Wl,-rpath=. -O2 -g")
SET(CMAKE_BUILD_TYPE "Release")
endif()

if(${BIT} STREQUAL "32")
    MESSAGE(STATUS "using 32bits (x86)")
    SET(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -m32")
    SET(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -m32")
    SET(QT5_GCC "$ENV{QT5GCC32}")
elseif(${BIT} STREQUAL "64")
    MESSAGE(STATUS "using 64bits (x86_64)")
    SET(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -m64")
    SET(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -m64")
    SET(QT5_GCC "$ENV{QT5GCC64}")
elseif(${BIT} STREQUAL "arm32")
    MESSAGE(STATUS "using arm32 (armhf)")
    SET(CMAKE_C_COMPILER "arm-linux-gnueabihf-gcc")
    SET(CMAKE_CXX_COMPILER "arm-linux-gnueabihf-g++")
elseif(${BIT} STREQUAL "arm64")
    MESSAGE(STATUS "using arm64 (aarch64)")
    SET(CMAKE_C_COMPILER "aarch64-linux-gnu-gcc")
    SET(CMAKE_CXX_COMPILER "aarch64-linux-gnu-g++")
else()
    MESSAGE(FATAL_ERROR "Unknown BIT value: ${BIT}. Please set BIT to '32', '64', 'arm32', or 'arm64'")
endif()

#建立项目
PROJECT(${PROJECT_NAME})

#MESSAGE("PROJECT_SOURCE_DIR = ${PROJECT_SOURCE_DIR}")
#${PROJECT_SOURCE_DIR}="xueyelingshuang/src/LumaPlayer/LumaPlayer"
SET(EXECUTABLE_OUTPUT_PATH ${PROJECT_SOURCE_DIR}/../../../lib/)
#SET(LIBRARY_OUTPUT_PATH ${PROJECT_SOURCE_DIR}/../../../lib/)

#添加包含目录
INCLUDE_DIRECTORIES(${PROJECT_SOURCE_DIR}/../../../include/)

#添加库目录
LINK_DIRECTORIES(${PROJECT_SOURCE_DIR}/../../../lib/)

SET(CMAKE_INCLUDE_CURRENT_DIR ON) 

#添加QT库
SET(CMAKE_AUTOMOC ON)
SET(CMAKE_AUTORCC ON)
SET(CMAKE_AUTOUIC ON)

SET(Qt5_DIR "${QT5_GCC}/lib/cmake/Qt5")

INCLUDE_DIRECTORIES("${QT5_GCC}/include/")

#添加Qt依赖库
SET(QTRELY Widgets;Core;Gui;Multimedia;Qml;Quick)

list(APPEND QTRELYLIST ${QTRELY})
foreach (EVERYQTRELY ${QTRELYLIST})
SET(Qt5${EVERYQTRELY}_DIR "${QT5_GCC}/lib/cmake/Qt5${EVERYQTRELY}")
FIND_PACKAGE(Qt5${EVERYQTRELY})
INCLUDE_DIRECTORIES("${QT5_GCC}/include/Qt${EVERYQTRELY}/")
endforeach ()

#查找当前文件夹中的所有源代码文件，也可以通过Set命令将所有文件设置为一个变量
AUX_SOURCE_DIRECTORY("${PROJECT_SOURCE_DIR}/" DIR_SRCS)
#查找当前文件夹中所有的ui文件
FILE(GLOB UI_FILES "${PROJECT_SOURCE_DIR}/*.ui")
#转换UI文件
QT5_WRAP_UI(WRAP_FILES ${UI_FILES})
#设置qrc
SET(RCC_FILES "${PROJECT_SOURCE_DIR}/LumaPlayer.qrc")

#编译可执行程序
ADD_EXECUTABLE(${PROJECT_NAME} ${DIR_SRCS} ${RCC_FILES} ${WRAP_FILES})
##如果要生成静态链接库
##ADD_LIBRARY(${PROJECT_NAME} STATIC ${DIR_SRCS} ${RCC_FILES} ${WRAP_FILES)
##如果要生成动态链接库
##ADD_LIBRARY(${PROJECT_NAME} SHARED ${DIR_SRCS} ${RCC_FILES} ${WRAP_FILES)
foreach (EVERYQTRELY ${QTRELYLIST})
TARGET_LINK_LIBRARIES(${PROJECT_NAME} "${QT5_GCC}/lib/libQt5${EVERYQTRELY}.so")
endforeach ()

#添加库
list(APPEND LIBRARYLIST ${LIBRARIESRELY})
foreach (EVERYLIBRARY ${LIBRARYLIST})
TARGET_LINK_LIBRARIES(${PROJECT_NAME} ${PROJECT_SOURCE_DIR}/../../../lib/${EVERYLIBRARY})
endforeach ()

# Generate native desktop icons from the single PNG source.
IF(APPLE OR UNIX)
    FIND_PROGRAM(LUMAPLAYER_ICON_SHELL sh)
    IF(NOT LUMAPLAYER_ICON_SHELL)
        MESSAGE(FATAL_ERROR "sh is required to generate application icons")
    ENDIF()
    SET(LUMAPLAYER_ICON_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/res/LumaPlayer.png")
    SET(LUMAPLAYER_ICON_SCRIPT "${CMAKE_CURRENT_SOURCE_DIR}/LumaPlayerIcon.sh")
    SET(LUMAPLAYER_ICON_DIR "${CMAKE_CURRENT_BINARY_DIR}/native-icons")
    IF(APPLE)
        SET(LUMAPLAYER_ICON_FORMAT ICNS)
        SET(LUMAPLAYER_NATIVE_ICON "${LUMAPLAYER_ICON_DIR}/LumaPlayer.icns")
        SET(LUMAPLAYER_ICON_OUTPUTS "${LUMAPLAYER_NATIVE_ICON}")
    ELSE()
        SET(LUMAPLAYER_ICON_FORMAT PNG)
        SET(LUMAPLAYER_NATIVE_ICON "${LUMAPLAYER_ICON_DIR}/${PROJECT_NAME}.png")
        SET(LUMAPLAYER_ICON_OUTPUTS "${LUMAPLAYER_NATIVE_ICON}"
            "${LUMAPLAYER_ICON_DIR}/${PROJECT_NAME}.desktop")
    ENDIF()
    ADD_CUSTOM_COMMAND(OUTPUT ${LUMAPLAYER_ICON_OUTPUTS}
        COMMAND "${LUMAPLAYER_ICON_SHELL}" "${LUMAPLAYER_ICON_SCRIPT}"
            --source "${LUMAPLAYER_ICON_SOURCE}" --output "${LUMAPLAYER_NATIVE_ICON}"
            --format ${LUMAPLAYER_ICON_FORMAT} --name "${PROJECT_NAME}"
        DEPENDS "${LUMAPLAYER_ICON_SOURCE}" "${LUMAPLAYER_ICON_SCRIPT}"
        VERBATIM)
    ADD_CUSTOM_TARGET(${PROJECT_NAME}_icon DEPENDS ${LUMAPLAYER_ICON_OUTPUTS})
    ADD_DEPENDENCIES(${PROJECT_NAME} ${PROJECT_NAME}_icon)
    IF(APPLE)
        SET_SOURCE_FILES_PROPERTIES("${LUMAPLAYER_NATIVE_ICON}" PROPERTIES
            GENERATED TRUE MACOSX_PACKAGE_LOCATION Resources)
        TARGET_SOURCES(${PROJECT_NAME} PRIVATE "${LUMAPLAYER_NATIVE_ICON}")
        SET_TARGET_PROPERTIES(${PROJECT_NAME} PROPERTIES
            MACOSX_BUNDLE TRUE MACOSX_BUNDLE_ICON_FILE LumaPlayer.icns)
    ELSE()
        INSTALL(TARGETS ${PROJECT_NAME} RUNTIME DESTINATION bin)
        INSTALL(FILES "${LUMAPLAYER_NATIVE_ICON}" DESTINATION share/pixmaps)
        INSTALL(FILES "${LUMAPLAYER_ICON_DIR}/${PROJECT_NAME}.desktop"
            DESTINATION share/applications)
    ENDIF()
ENDIF()
