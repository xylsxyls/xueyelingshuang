cmake_minimum_required(VERSION 3.15)
if(BIT STREQUAL "wasm64")
    string(APPEND CMAKE_C_FLAGS_INIT " -sMEMORY64=1")
    string(APPEND CMAKE_CXX_FLAGS_INIT " -sMEMORY64=1")
    string(APPEND CMAKE_EXE_LINKER_FLAGS_INIT " -sMEMORY64=1")
endif()
project(CSystemTest LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(root "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
include("${root}/src/PlatformTest/PlatformTest/src/PlatformTest.cmake")
if(NOT SDK_ROOT)
    set(SDK_ROOT "$ENV{CLOUD_REBUILD_SDK_ROOT}")
endif()
if(NOT SDK_ROOT)
    set(SDK_ROOT "${root}")
endif()
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:DebugDLL>")
endif()
add_executable(CSystemTest CSystemTest.cpp CSystemRegressionTests.cpp)
platform_test_link(CSystemTest "${SDK_ROOT}")
option(PROCESS_WORK_PORTABLE_IDENTITY "Match a desktop SDK built with shared directory identity" OFF)
if(PROCESS_WORK_PORTABLE_IDENTITY)
    target_compile_definitions(CSystemTest PRIVATE PROCESS_WORK_PORTABLE_IDENTITY)
endif()
if(EMSCRIPTEN)
    platform_test_wasm_headers(CSystemTest "${SDK_ROOT}" CSystem)
    math(EXPR wasm_bits "${CMAKE_SIZEOF_VOID_P} * 8")
    if(BIT AND NOT BIT STREQUAL "wasm${wasm_bits}")
        message(FATAL_ERROR "BIT disagrees with the compiler pointer size")
    endif()
    if(DEBUGRELEASE STREQUAL "debug" OR CMAKE_BUILD_TYPE STREQUAL "Debug")
        set(suffix d)
    endif()
    target_link_libraries(CSystemTest PRIVATE "${SDK_ROOT}/lib/CSystem${suffix}.a")
    target_compile_definitions(CSystemTest PRIVATE STATIC_LIB)
    target_compile_options(CSystemTest PRIVATE -fexceptions)
    target_link_options(CSystemTest PRIVATE -fexceptions -sALLOW_MEMORY_GROWTH=1)
else()
    if(MSVC)
        target_link_directories(CSystemTest PRIVATE "${SDK_ROOT}/lib")
        target_link_libraries(CSystemTest PRIVATE "${SDK_ROOT}/lib/CSystem$<$<CONFIG:Debug>:d>.lib")
    else()
        platform_test_sdk_library(prepared_library CSystem)
        target_link_libraries(CSystemTest PRIVATE "${prepared_library}")
    endif()
    if(APPLE)
        find_library(security_framework Security)
        target_link_libraries(CSystemTest PRIVATE "${security_framework}")
        if(CMAKE_SYSTEM_NAME STREQUAL "Darwin")
            find_library(proc_library proc)
            target_link_libraries(CSystemTest PRIVATE "${proc_library}")
        endif()
    elseif(UNIX)
        target_link_libraries(CSystemTest PRIVATE ${CMAKE_DL_LIBS})
    endif()
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
        target_link_libraries(CSystemTest PRIVATE rt)
    endif()
    foreach(dependency IN LISTS LIBRARIESRELY)
        if(IS_ABSOLUTE "${dependency}")
            target_link_libraries(CSystemTest PRIVATE "${dependency}")
        else()
            target_link_libraries(CSystemTest PRIVATE "${SDK_ROOT}/lib/${dependency}")
        endif()
    endforeach()
    if(UNIX)
        find_package(Threads REQUIRED)
        target_link_libraries(CSystemTest PRIVATE Threads::Threads)
    endif()
endif()
enable_testing()
add_library_platform_test(CSystemTest TAKES_ARGS)
