cmake_minimum_required(VERSION 3.15)
if(BIT STREQUAL "wasm64")
    string(APPEND CMAKE_C_FLAGS " -sMEMORY64=1")
    string(APPEND CMAKE_CXX_FLAGS " -sMEMORY64=1")
    string(APPEND CMAKE_EXE_LINKER_FLAGS " -sMEMORY64=1")
elseif(BIT MATCHES "^(32|64)$" AND CMAKE_HOST_SYSTEM_NAME STREQUAL "Linux" AND NOT CMAKE_TOOLCHAIN_FILE
    AND (NOT CMAKE_SYSTEM_NAME OR CMAKE_SYSTEM_NAME STREQUAL "Linux"))
    # 使桌面 Linux 的测试程序与 version 脚本准备的 x86 SDK 使用相同位宽
    cmake_host_system_information(RESULT host_cpu QUERY OS_PLATFORM)
    if(NOT host_cpu MATCHES "^(x86_64|amd64|i[3-6]86|x86)$")
        message(FATAL_ERROR "Choose arm32/arm64 or provide the requested x86 toolchain")
    endif()
    string(APPEND CMAKE_CXX_FLAGS_INIT " -m${BIT}")
endif()
include("${CMAKE_CURRENT_SOURCE_DIR}/../../PlatformTest/PlatformTest/src/PlatformTest.cmake")
platform_test_select_linux_arm()
project(CDumpTest LANGUAGES CXX)
if(BIT MATCHES "^(32|64|arm32|arm64)$")
    math(EXPR test_bits "${CMAKE_SIZEOF_VOID_P} * 8")
    if(NOT BIT MATCHES "^(arm)?${test_bits}$")
        message(FATAL_ERROR "BIT=${BIT} disagrees with the ${test_bits}-bit test compiler; use a matching build directory and toolchain")
    endif()
endif()
include(CTest)
# 注册是进程级一次性操作，本工程没有可供 CTest 调用的 --benchmark 入口
set(LIBRARY_TEST_BENCHMARKS OFF)
build_sdk_test(CDumpTest PROJECTS CDump)
