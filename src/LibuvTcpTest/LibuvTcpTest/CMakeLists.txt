cmake_minimum_required(VERSION 3.16)

# 非 Windows 使用目标工具链；Windows 仍由 version 准备依赖、VS 构建。
set(BIT "64" CACHE STRING "32, 64, arm32, arm64")
set(DEBUGRELEASE "release" CACHE STRING "debug or release")
if(NOT BIT MATCHES "^(32|64|arm32|arm64)$" OR NOT DEBUGRELEASE MATCHES "^(debug|release)$")
    message(FATAL_ERROR "Invalid BIT or DEBUGRELEASE")
endif()
if(DEBUGRELEASE STREQUAL "debug")
    set(CMAKE_BUILD_TYPE Debug CACHE STRING "Build configuration" FORCE)
else()
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build configuration" FORCE)
endif()
project(LibuvTcpTest LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# 不把宿主 x86 编译器当成 ARM/Android/iOS 交叉编译器。ABI 由调用者的工具链选择。
include(CheckCXXSourceCompiles)
if(BIT STREQUAL "32")
    set(test_arch "defined(__i386__) || defined(_M_IX86)")
elseif(BIT STREQUAL "64")
    set(test_arch "defined(__x86_64__) || defined(_M_X64)")
elseif(BIT STREQUAL "arm32")
    set(test_arch "defined(__arm__) || defined(_M_ARM)")
else()
    set(test_arch "defined(__aarch64__) || defined(_M_ARM64)")
endif()
set(test_saved_try_compile "${CMAKE_TRY_COMPILE_TARGET_TYPE}")
set(CMAKE_TRY_COMPILE_TARGET_TYPE STATIC_LIBRARY)
unset(LIBUV_TEST_ABI_MATCHES CACHE)
check_cxx_source_compiles("#if !(${test_arch})\n#error Wrong target architecture\n#endif\nint targetCheck() { return 0; }" LIBUV_TEST_ABI_MATCHES)
if(test_saved_try_compile)
    set(CMAKE_TRY_COMPILE_TARGET_TYPE "${test_saved_try_compile}")
else()
    unset(CMAKE_TRY_COMPILE_TARGET_TYPE)
endif()
if(NOT LIBUV_TEST_ABI_MATCHES)
    message(FATAL_ERROR "BIT disagrees with the selected toolchain")
endif()

get_filename_component(test_repository "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(test_sdk "${test_repository}")
if(DEFINED ENV{CLOUD_REBUILD_SDK_ROOT} AND NOT "$ENV{CLOUD_REBUILD_SDK_ROOT}" STREQUAL "")
    set(test_sdk "$ENV{CLOUD_REBUILD_SDK_ROOT}")
endif()
set(SDK_ROOT "${test_sdk}" CACHE PATH "Prepared target SDK")
file(GLOB test_sources CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/*.cpp")
add_executable(LibuvTcpTest ${test_sources})
option(LIBUV_TEST_BENCHMARK "Build performance entry without allocation-failure hooks" OFF)
if(LIBUV_TEST_BENCHMARK)
    target_compile_definitions(LibuvTcpTest PRIVATE LIBUV_TEST_BENCHMARK)
endif()
set_target_properties(LibuvTcpTest PROPERTIES DEBUG_POSTFIX d)
target_compile_definitions(LibuvTcpTest PRIVATE STATIC_LIB WIN32_LEAN_AND_MEAN)
target_include_directories(LibuvTcpTest PRIVATE "${SDK_ROOT}/include")

# 白盒测试链接静态库。这里仅链接 version 已编译的目标依赖，不触发二次构建。
if(NOT LIBRARIESRELY)
    message(FATAL_ERROR "Supply LIBRARIESRELY with prepared LibuvTcp, CTaskThreadManager, Semaphore and libuv archives")
endif()
set(test_libraries)
foreach(test_library IN LISTS LIBRARIESRELY)
    if(NOT IS_ABSOLUTE "${test_library}")
        set(test_library "${SDK_ROOT}/lib/${test_library}")
    endif()
    if(NOT EXISTS "${test_library}" OR IS_DIRECTORY "${test_library}")
        message(FATAL_ERROR "Missing target library: ${test_library}")
    endif()
    list(APPEND test_libraries "${test_library}")
endforeach()
if(CMAKE_SYSTEM_NAME MATCHES "^(Linux|Android)$")
    target_link_libraries(LibuvTcpTest PRIVATE "-Wl,--start-group" ${test_libraries} "-Wl,--end-group")
else()
    target_link_libraries(LibuvTcpTest PRIVATE ${test_libraries})
endif()
find_package(Threads REQUIRED)
target_link_libraries(LibuvTcpTest PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_link_libraries(LibuvTcpTest PRIVATE rt atomic)
elseif(WIN32)
    target_link_libraries(LibuvTcpTest PRIVATE ws2_32 iphlpapi userenv psapi)
endif()
if(APPLE)
    set_target_properties(LibuvTcpTest PROPERTIES BUILD_RPATH "@loader_path")
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    set_target_properties(LibuvTcpTest PROPERTIES BUILD_RPATH "$ORIGIN")
endif()
