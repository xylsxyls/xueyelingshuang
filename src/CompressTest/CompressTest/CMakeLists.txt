cmake_minimum_required(VERSION 3.15)

# Memory64需要在编译器探测前选择指针ABI。
if(BIT STREQUAL "wasm64")
    string(APPEND CMAKE_C_FLAGS_INIT " -sMEMORY64=1")
    string(APPEND CMAKE_CXX_FLAGS_INIT " -sMEMORY64=1")
    string(APPEND CMAKE_EXE_LINKER_FLAGS_INIT " -sMEMORY64=1")
endif()
project(CompressTest LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND BIT MATCHES "^arm(32|64)$")
    # 交叉测试须使用目标工具链，单独传入 BIT 不会切换编译器。
    if(BIT STREQUAL "arm32")
        set(arm_test_abi "defined(__arm__) && __ARM_ARCH >= 7 && __ARM_ARCH_PROFILE == 'A' && defined(__ARM_PCS_VFP) && __SIZEOF_POINTER__ == 4")
    else()
        set(arm_test_abi "defined(__aarch64__) && __SIZEOF_POINTER__ == 8")
    endif()
    include(CheckCXXSourceCompiles)
    unset(COMPRESS_TEST_ARM_ABI_MATCHES CACHE)
    check_cxx_source_compiles("#if !(${arm_test_abi}) || __BYTE_ORDER__ != __ORDER_LITTLE_ENDIAN__\n#error Wrong Linux ARM ABI\n#endif\nint main() { return 0; }" COMPRESS_TEST_ARM_ABI_MATCHES)
    if(NOT COMPRESS_TEST_ARM_ABI_MATCHES)
        message(FATAL_ERROR "The Linux test toolchain does not match BIT=${BIT} and the supplied little-endian ARM SDK")
    endif()
endif()
get_filename_component(root "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
include("${root}/src/PlatformTest/PlatformTest/src/PlatformTest.cmake")
if(NOT SDK_ROOT)
    set(SDK_ROOT "$ENV{CLOUD_REBUILD_SDK_ROOT}")
endif()
if(NOT SDK_ROOT)
    set(SDK_ROOT "${root}")
endif()
# Emscripten 的链接响应文件需要 CMake 路径分隔符，包括从 Windows 环境变量取得的 SDK 路径。
get_filename_component(SDK_ROOT "${SDK_ROOT}" ABSOLUTE)

add_executable(CompressTest CompressTest.cpp)
platform_test_link(CompressTest "${SDK_ROOT}")
if(EMSCRIPTEN)
    platform_test_wasm_headers(CompressTest "${SDK_ROOT}" Compress)
    math(EXPR wasm_bits "${CMAKE_SIZEOF_VOID_P} * 8")
    if(BIT AND NOT BIT STREQUAL "wasm${wasm_bits}")
        message(FATAL_ERROR "BIT and the Emscripten compiler pointer size disagree")
    endif()
    set(suffix "$<$<CONFIG:Debug>:d>")
    if(NOT CMAKE_CONFIGURATION_TYPES AND NOT CMAKE_BUILD_TYPE AND DEBUGRELEASE STREQUAL "debug")
        set(suffix d)
    endif()
    target_compile_definitions(CompressTest PRIVATE STATIC_LIB)
    target_compile_options(CompressTest PRIVATE -fexceptions)
    target_link_options(CompressTest PRIVATE -fexceptions -sALLOW_MEMORY_GROWTH=1)
    target_link_libraries(CompressTest PRIVATE "${SDK_ROOT}/lib/Compress${suffix}.a" "${SDK_ROOT}/lib/libzstd.a")
else()
    platform_test_sdk_library(prepared_library Compress)
    target_link_libraries(CompressTest PRIVATE "${prepared_library}")
    if(MSVC)
        set_property(TARGET CompressTest PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:DebugDLL>")
        target_link_directories(CompressTest PRIVATE "${SDK_ROOT}/lib")
        target_link_libraries(CompressTest PRIVATE "${SDK_ROOT}/lib/zstd$<$<CONFIG:Debug>:d>.lib")
    else()
        target_link_libraries(CompressTest PRIVATE "${SDK_ROOT}/lib/libzstd.a")
    endif()
endif()
include(CTest)
if(BUILD_TESTING)
    add_library_platform_test(CompressTest TAKES_ARGS)
endif()
