cmake_minimum_required(VERSION 3.15)

# Memory64需要在编译器探测前选择指针ABI。
if(BIT STREQUAL "wasm64")
    string(APPEND CMAKE_C_FLAGS_INIT " -sMEMORY64=1")
    string(APPEND CMAKE_CXX_FLAGS_INIT " -sMEMORY64=1")
    string(APPEND CMAKE_EXE_LINKER_FLAGS_INIT " -sMEMORY64=1")
endif()
project(CompressTest LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(root "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
if(NOT SDK_ROOT)
    set(SDK_ROOT "$ENV{CLOUD_REBUILD_SDK_ROOT}")
endif()
if(NOT SDK_ROOT)
    set(SDK_ROOT "${root}")
endif()

# 使用生产源码和before准备的当前目标SDK，测试无需依赖宿主机安装的库。
set(compress_source "${root}/src/Compress/Compress/src")
if(EMSCRIPTEN)
    math(EXPR wasm_bits "${CMAKE_SIZEOF_VOID_P} * 8")
    if(BIT AND NOT BIT STREQUAL "wasm${wasm_bits}")
        message(FATAL_ERROR "BIT and the Emscripten compiler pointer size disagree")
    endif()
    set(vendor "${SDK_ROOT}/lib/libzstd.a")
elseif(MSVC)
    math(EXPR win_bits "${CMAKE_SIZEOF_VOID_P} * 8")
    set(vendor "${SDK_ROOT}/lib/zstd$<$<CONFIG:Debug>:d>.lib")
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:DebugDLL>")
else()
    include("${root}/common/CloudRebuild/NativeVendor.cmake")
    native_vendor_abi(native_abi)
    set(vendor "${SDK_ROOT}/lib/libzstd.a")
endif()
add_library(Compress STATIC "${compress_source}/Compress.cpp")
set_target_properties(Compress PROPERTIES DEBUG_POSTFIX d)
target_include_directories(Compress PUBLIC "${compress_source}" PRIVATE "${SDK_ROOT}/include")
target_compile_definitions(Compress PUBLIC STATIC_LIB PRIVATE COMPRESS_NO_AUTOLINK)
target_link_libraries(Compress PUBLIC "${vendor}")
add_executable(CompressTest CompressTest.cpp)
target_link_libraries(CompressTest PRIVATE Compress)
if(EMSCRIPTEN)
    target_link_options(CompressTest PRIVATE -sALLOW_MEMORY_GROWTH=1)
endif()
include(CTest)
if(BUILD_TESTING)
    include("${root}/common/wasm/LibraryTestHost.cmake")
    add_library_platform_test(CompressTest TAKES_ARGS)
endif()
