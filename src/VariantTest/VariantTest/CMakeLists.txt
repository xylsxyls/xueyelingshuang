# 单配置生成器用同一配置编译测试并选择 SDK；多配置生成器按构建时的配置选择。
if(DEFINED DEBUGRELEASE AND NOT DEBUGRELEASE STREQUAL "")
    string(TOLOWER "${DEBUGRELEASE}" sdk_mode)
    if(sdk_mode STREQUAL "debug")
        set(requested_config Debug)
    elseif(sdk_mode STREQUAL "release")
        set(requested_config Release)
    else()
        message(FATAL_ERROR "DEBUGRELEASE must be debug or release")
    endif()
    get_property(multi_config GLOBAL PROPERTY GENERATOR_IS_MULTI_CONFIG)
    if(NOT multi_config)
        if(NOT CMAKE_BUILD_TYPE)
            set(CMAKE_BUILD_TYPE "${requested_config}" CACHE STRING "Build configuration" FORCE)
        else()
            string(TOLOWER "${CMAKE_BUILD_TYPE}" build_mode)
            if((build_mode STREQUAL "debug" AND NOT sdk_mode STREQUAL "debug") OR
                (NOT build_mode STREQUAL "debug" AND sdk_mode STREQUAL "debug"))
                message(FATAL_ERROR "DEBUGRELEASE conflicts with CMAKE_BUILD_TYPE")
            endif()
        endif()
    endif()
endif()

# 相同的指针宽度也可能对应不同的指令集。检查编译器的目标宏，交叉编译时无需运行探针。
function(variant_test_check_arch)
    if(BIT STREQUAL "32")
        set(arch_check "defined(__i386__) || defined(_M_IX86)")
    elseif(BIT STREQUAL "64")
        set(arch_check "defined(__x86_64__) || defined(_M_X64)")
    elseif(BIT STREQUAL "arm32")
        set(arch_check "defined(__arm__) || defined(_M_ARM)")
    elseif(BIT STREQUAL "arm64")
        set(arch_check "defined(__aarch64__) || defined(_M_ARM64)")
    else()
        message(FATAL_ERROR "Native BIT must be 32, 64, arm32 or arm64")
    endif()
    math(EXPR pointer_bits "${CMAKE_SIZEOF_VOID_P} * 8")
    if(NOT BIT MATCHES "${pointer_bits}$")
        message(FATAL_ERROR "BIT disagrees with the compiler pointer size")
    endif()
    include(CheckCXXSourceCompiles)
    set(CMAKE_TRY_COMPILE_TARGET_TYPE STATIC_LIBRARY)
    unset(VARIANT_TEST_MATCHING_ARCH CACHE)
    check_cxx_source_compiles("#if !(${arch_check})\n#error Unexpected CPU\n#endif\nint main() { return 0; }" VARIANT_TEST_MATCHING_ARCH)
    if(NOT VARIANT_TEST_MATCHING_ARCH)
        message(FATAL_ERROR "BIT disagrees with the compiler target architecture")
    endif()
endfunction()
if(NOT EMSCRIPTEN AND DEFINED BIT AND NOT BIT STREQUAL "")
    variant_test_check_arch()
endif()
