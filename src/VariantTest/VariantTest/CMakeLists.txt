cmake_minimum_required(VERSION 3.15)
if(BIT STREQUAL "wasm64")
    string(APPEND CMAKE_C_FLAGS_INIT " -sMEMORY64=1")
    string(APPEND CMAKE_CXX_FLAGS_INIT " -sMEMORY64=1")
    string(APPEND CMAKE_EXE_LINKER_FLAGS_INIT " -sMEMORY64=1")
endif()
project(VariantTest LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(root "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
include("${root}/src/PlatformTest/PlatformTest/src/PlatformTest.cmake")
if(NOT SDK_ROOT)
    set(SDK_ROOT "$ENV{CLOUD_REBUILD_SDK_ROOT}")
endif()
if(NOT SDK_ROOT)
    set(SDK_ROOT "${root}")
endif()
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:DebugDLL>")
endif()
add_executable(VariantTest VariantTest.cpp VariantScenarios.cpp)
platform_test_link(VariantTest "${SDK_ROOT}")
target_include_directories(VariantTest PRIVATE "${SDK_ROOT}/include")
if(EMSCRIPTEN)
    platform_test_wasm_headers(VariantTest "${SDK_ROOT}" Variant)
    math(EXPR wasm_bits "${CMAKE_SIZEOF_VOID_P} * 8")
    if(BIT AND NOT BIT STREQUAL "wasm${wasm_bits}")
        message(FATAL_ERROR "BIT disagrees with the compiler pointer size")
    endif()
    if(DEBUGRELEASE STREQUAL "debug" OR CMAKE_BUILD_TYPE STREQUAL "Debug")
        set(suffix d)
    endif()
    target_link_libraries(VariantTest PRIVATE "${SDK_ROOT}/lib/Variant${suffix}.a")
    target_compile_definitions(VariantTest PRIVATE STATIC_LIB)
    target_compile_options(VariantTest PRIVATE -fexceptions)
    target_link_options(VariantTest PRIVATE -fexceptions -sALLOW_MEMORY_GROWTH=1)
else()
    if(MSVC)
        target_link_directories(VariantTest PRIVATE "${SDK_ROOT}/lib")
    endif()
    foreach(library IN ITEMS Variant)
        platform_test_sdk_library(prepared_library "${library}")
        target_link_libraries(VariantTest PRIVATE "${prepared_library}")
    endforeach()
    foreach(dependency IN LISTS LIBRARIESRELY)
        if(IS_ABSOLUTE "${dependency}")
            target_link_libraries(VariantTest PRIVATE "${dependency}")
        else()
            target_link_libraries(VariantTest PRIVATE "${SDK_ROOT}/lib/${dependency}")
        endif()
    endforeach()
    if(UNIX)
        find_package(Threads REQUIRED)
        target_link_libraries(VariantTest PRIVATE Threads::Threads)
    endif()
endif()
enable_testing()
add_library_platform_test(VariantTest TAKES_ARGS)
