cmake_minimum_required(VERSION 3.15)
project(HiRedisTest LANGUAGES CXX)
if(NOT WIN32 AND NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "HiRedisTest supports Windows and Linux")
endif()
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(root "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
include("${root}/src/PlatformTest/PlatformTest/src/PlatformTest.cmake")
if(NOT SDK_ROOT)
    set(SDK_ROOT "$ENV{CLOUD_REBUILD_SDK_ROOT}")
endif()
if(NOT SDK_ROOT)
    set(SDK_ROOT "${root}")
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    foreach(required IN ITEMS include/hiredisinclude/hiredis.h lib/libhiredis.a)
        if(NOT EXISTS "${SDK_ROOT}/${required}")
            message(FATAL_ERROR "Missing prepared Linux SDK file: ${SDK_ROOT}/${required}")
        endif()
    endforeach()
    if(NOT LIBRARIESRELY)
        set(LIBRARIESRELY libhiredis.a)
    endif()
endif()
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:DebugDLL>")
endif()
add_executable(HiRedisTest HiRedisTest.cpp)
platform_test_link(HiRedisTest "${SDK_ROOT}")
target_include_directories(HiRedisTest PRIVATE "${SDK_ROOT}/include")
if(MSVC)
    target_link_directories(HiRedisTest PRIVATE "${SDK_ROOT}/lib")
endif()
foreach(library IN ITEMS HiRedis)
    platform_test_sdk_library(prepared_library "${library}")
    target_link_libraries(HiRedisTest PRIVATE "${prepared_library}")
endforeach()
foreach(dependency IN LISTS LIBRARIESRELY)
    if(IS_ABSOLUTE "${dependency}")
        target_link_libraries(HiRedisTest PRIVATE "${dependency}")
    else()
        target_link_libraries(HiRedisTest PRIVATE "${SDK_ROOT}/lib/${dependency}")
    endif()
endforeach()
if(UNIX)
    find_package(Threads REQUIRED)
    target_link_libraries(HiRedisTest PRIVATE Threads::Threads)
endif()
enable_testing()
add_test(NAME HiRedisTest COMMAND HiRedisTest)
