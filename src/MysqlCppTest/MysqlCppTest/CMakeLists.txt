cmake_minimum_required(VERSION 3.15)
if(BIT STREQUAL "wasm64")
    string(APPEND CMAKE_C_FLAGS_INIT " -sMEMORY64=1")
    string(APPEND CMAKE_CXX_FLAGS_INIT " -sMEMORY64=1")
    string(APPEND CMAKE_EXE_LINKER_FLAGS_INIT " -sMEMORY64=1")
endif()
project(MysqlCppTest LANGUAGES C CXX)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Linux SDK for mysqlcpp is pending; no Linux release is packaged for this target")
endif()
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(root "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
if(NOT SDK_ROOT)
    set(SDK_ROOT "$ENV{CLOUD_REBUILD_SDK_ROOT}")
endif()
if(NOT SDK_ROOT)
    set(SDK_ROOT "${root}")
endif()
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:DebugDLL>")
endif()
add_executable(MysqlCppTest MysqlCppTest.cpp "${root}/src/MysqlCpp/MysqlCpp/src/MysqlCppHelper.cpp")
target_include_directories(MysqlCppTest PRIVATE "${SDK_ROOT}/include/mysql-connector-cpp" "${SDK_ROOT}/include")
target_compile_definitions(MysqlCppTest PRIVATE CPPCONN_LIB_BUILD)
if(EMSCRIPTEN)
    message(FATAL_ERROR "This suite requires a native target")
else()
    include("${root}/common/PlatformTest/PreparedLibrary.cmake")
    if(MSVC)
        target_link_directories(MysqlCppTest PRIVATE "${SDK_ROOT}/lib")
    endif()
    foreach(library IN ITEMS MysqlCpp CStringManager)
        platform_test_sdk_library(prepared_library "${library}")
        target_link_libraries(MysqlCppTest PRIVATE "${prepared_library}")
    endforeach()
    foreach(dependency IN LISTS LIBRARIESRELY)
        if(IS_ABSOLUTE "${dependency}")
            target_link_libraries(MysqlCppTest PRIVATE "${dependency}")
        else()
            target_link_libraries(MysqlCppTest PRIVATE "${SDK_ROOT}/lib/${dependency}")
        endif()
    endforeach()
    if(UNIX)
        find_package(Threads REQUIRED)
        target_link_libraries(MysqlCppTest PRIVATE Threads::Threads)
    endif()
endif()
if(WIN32)
    add_custom_command(TARGET MysqlCppTest POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "${SDK_ROOT}/lib/libmysql.dll" "$<TARGET_FILE_DIR:MysqlCppTest>/libmysql.dll")
endif()
enable_testing()
add_test(NAME MysqlCppTest COMMAND MysqlCppTest)
