cmake_minimum_required(VERSION 3.15)
get_filename_component(root "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
include("${root}/src/PlatformTest/PlatformTest/src/PlatformTest.cmake")
platform_test_select_linux_arm()
if(CMAKE_HOST_SYSTEM_NAME STREQUAL "Linux" AND BIT MATCHES "^(32|64)$" AND NOT CMAKE_TOOLCHAIN_FILE)
    cmake_host_system_information(RESULT host_cpu QUERY OS_PLATFORM)
    string(TOLOWER "${host_cpu}" host_cpu)
    if(host_cpu MATCHES "^(x86_64|amd64|x86|i[3-6]86)$")
        string(APPEND CMAKE_CXX_FLAGS_INIT " -m${BIT}")
    endif()
endif()
project(MysqlCppTest LANGUAGES CXX)
if(NOT WIN32 AND NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "MysqlCppTest supports Windows and Linux")
endif()
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT SDK_ROOT)
    set(SDK_ROOT "$ENV{CLOUD_REBUILD_SDK_ROOT}")
endif()
if(NOT SDK_ROOT)
    set(SDK_ROOT "${root}")
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    if(DEFINED BIT AND NOT BIT STREQUAL "")
        if(BIT STREQUAL "32")
            set(required_abi "defined(__i386__) && __SIZEOF_POINTER__ == 4")
        elseif(BIT STREQUAL "64")
            set(required_abi "defined(__x86_64__) && __SIZEOF_POINTER__ == 8")
        elseif(BIT STREQUAL "arm32")
            set(required_abi "defined(__arm__) && defined(__ARM_PCS_VFP) && __SIZEOF_POINTER__ == 4")
        elseif(BIT STREQUAL "arm64")
            set(required_abi "defined(__aarch64__) && __SIZEOF_POINTER__ == 8")
        else()
            message(FATAL_ERROR "Unsupported Linux BIT=${BIT}. Use 32, 64, arm32 or arm64.")
        endif()
        include(CheckCXXSourceCompiles)
        unset(mysqlcpp_test_abi_matches CACHE)
        check_cxx_source_compiles("#if !(${required_abi})\n#error Unexpected target ABI\n#endif\nint main() { return 0; }" mysqlcpp_test_abi_matches)
        if(NOT mysqlcpp_test_abi_matches)
            message(FATAL_ERROR "MysqlCppTest requires a Linux ${BIT} compiler and matching SDK. Use a separate build directory for each ABI.")
        endif()
    endif()
    foreach(required IN ITEMS include/mysql-connector-cpp/cppconn/driver.h lib/libmysqlcppconn-static.a lib/libmysqlclient.so)
        if(NOT EXISTS "${SDK_ROOT}/${required}")
            message(FATAL_ERROR "Missing prepared Linux SDK file: ${SDK_ROOT}/${required}")
        endif()
    endforeach()
    if(NOT LIBRARIESRELY)
        set(LIBRARIESRELY libmysqlcppconn-static.a libmysqlclient.so)
    endif()
endif()
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:DebugDLL>")
endif()
add_executable(MysqlCppTest MysqlCppTest.cpp)
platform_test_link(MysqlCppTest "${SDK_ROOT}")
target_include_directories(MysqlCppTest PRIVATE "${SDK_ROOT}/include/mysql-connector-cpp" "${SDK_ROOT}/include")
target_compile_definitions(MysqlCppTest PRIVATE CPPCONN_LIB_BUILD)
if(MSVC)
    target_link_directories(MysqlCppTest PRIVATE "${SDK_ROOT}/lib")
endif()
# 白盒用例使用 SDK 静态 MysqlCpp 中的内部实现。其他库仍按各自 SDK 类型查找。
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    set(saved_library_suffixes "${CMAKE_FIND_LIBRARY_SUFFIXES}")
    set(CMAKE_FIND_LIBRARY_SUFFIXES "${CMAKE_STATIC_LIBRARY_SUFFIX}")
endif()
platform_test_sdk_library(mysql_library MysqlCpp)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    set(CMAKE_FIND_LIBRARY_SUFFIXES "${saved_library_suffixes}")
endif()
platform_test_sdk_library(string_library CStringManager)
target_link_libraries(MysqlCppTest PRIVATE "${mysql_library}" "${string_library}")
foreach(dependency IN LISTS LIBRARIESRELY)
    if(IS_ABSOLUTE "${dependency}")
        target_link_libraries(MysqlCppTest PRIVATE "${dependency}")
    else()
        target_link_libraries(MysqlCppTest PRIVATE "${SDK_ROOT}/lib/${dependency}")
    endif()
endforeach()
if(UNIX)
    find_package(Threads REQUIRED)
    target_link_libraries(MysqlCppTest PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
endif()
if(WIN32)
    # 同一位宽的 SDK 可能只包含本配置的 DLL，也可将 PlatformTest 作为静态库分发。
    file(GLOB platform_test_runtimes "${SDK_ROOT}/lib/PlatformTest.dll" "${SDK_ROOT}/lib/PlatformTestd.dll")
    add_custom_command(TARGET MysqlCppTest POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "${SDK_ROOT}/lib/libmysql.dll" ${platform_test_runtimes} "$<TARGET_FILE_DIR:MysqlCppTest>" VERBATIM)
endif()
enable_testing()
add_test(NAME MysqlCppTest COMMAND MysqlCppTest)
set_tests_properties(MysqlCppTest PROPERTIES TIMEOUT 180)
