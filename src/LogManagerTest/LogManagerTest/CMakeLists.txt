cmake_minimum_required(VERSION 3.15)
if(BIT STREQUAL "wasm64")
    string(APPEND CMAKE_C_FLAGS " -sMEMORY64=1")
    string(APPEND CMAKE_CXX_FLAGS " -sMEMORY64=1")
    string(APPEND CMAKE_EXE_LINKER_FLAGS " -sMEMORY64=1")
endif()
include("${CMAKE_CURRENT_SOURCE_DIR}/../../PlatformTest/PlatformTest/src/PlatformTest.cmake")
platform_test_select_linux_arm()
project(LogManagerTest LANGUAGES CXX)
include(CTest)
if(MSVC)
    set(system_libraries ws2_32 iphlpapi psapi userenv bcrypt shell32 ole32 oleaut32 uuid gdi32 user32 dbghelp advapi32 crypt32 secur32 normaliz wldap32)
else()
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
        list(APPEND system_libraries rt)
    elseif(APPLE)
        list(APPEND system_libraries "-framework Security")
    endif()
endif()
build_sdk_test(LogManagerTest PROJECTS LogManager CSystem IntDateTime CStringManager
    SYSTEM_LIBRARIES ${system_libraries})
target_compile_definitions(LogManagerTest PRIVATE "$<$<CONFIG:Debug>:_DEBUG>")
if(WIN32)
    target_sources(LogManagerTest PRIVATE LogManagerArchiveTest.cpp)
endif()
if(EMSCRIPTEN)
    target_link_options(LogManagerTest PRIVATE -lnodefs.js)
endif()
